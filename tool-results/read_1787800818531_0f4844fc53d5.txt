   174→function isNewProduct(createdAt: string | null | undefined): boolean {
   175→  if (!createdAt) return false
   176→  const created = new Date(createdAt)
   177→  const thirtyDaysAgo = new Date()
   178→  thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30)
   179→  return created >= thirtyDaysAgo
   180→}

   191→// ==================== ANIMATED COUNTER ====================

   193→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   194→  const [count, setCount] = useState(0)
   195→  const ref = useRef(null)
   196→  const isInView = useInView(ref, { once: true })
   197→
   198→  useEffect(() => {
   199→    if (!isInView) return
   200→    let start = 0
   201→    const duration = 2000
   202→    const step = target / (duration / 16)
   203→    const timer = setInterval(() => {
   204→      start += step
   205→      if (start >= target) {
   206→        setCount(target)
   207→        clearInterval(timer)
   208→      } else {
   209→        setCount(Math.floor(start))
   210→      }
   211→    }, 16)
   212→    return () => clearInterval(timer)
   213→  }, [isInView, target])
   214→
   215→  return <span ref={ref}>{count}{suffix}</span>
   216→}

   220→function FadeInSection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   221→  const ref = useRef(null)
   222→  const isInView = useInView(ref, { once: true, margin: '-80px' })
   223→
   224→  return (
   225→    <motion.div
   226→      ref={ref}
   227→      initial={{ opacity: 0, y: 40 }}
   228→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   229→      transition={{ duration: 0.6, ease: 'easeOut' }}
   230→      className={className}
   231→    >
   232→      {children}
   233→    </motion.div>
   234→  )
   235→}
