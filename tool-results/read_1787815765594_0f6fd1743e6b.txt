   102→function getCategoryBorderColor(slug: string): string {
   103→  switch (slug) {
   104→    case 'savon-liquide':
   105→      return 'border-l-emerald-500'
   106→    case 'detergent':
   107→      return 'border-l-amber-500'
   108→    case 'eau-de-javel':
   109→      return 'border-l-cyan-500'
   110→    default:
   111→      return 'border-l-emerald-500'
   112→  }

   146→function OrderProgressIndicator({ step }: { step: 1 | 2 | 3 }) {
   147→  const steps = [
   148→    { num: 1, label: 'Informations' },
   149→    { num: 2, label: 'Livraison' },
   150→    { num: 3, label: 'Confirmation' },
   151→  ]
   152→  return (
   153→    <div className="flex items-center justify-between mb-6">
   154→      {steps.map((s, i) => (
   155→        <div key={s.num} className="flex items-center flex-1 last:flex-none">
   156→          <div className="flex flex-col items-center gap-1.5">
   157→            <div
   158→              className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-semibold transition-colors duration-300 ${
   159→                step >= s.num
   160→                  ? 'bg-emerald-600 text-white shadow-md shadow-emerald-600/25'
   161→                  : 'bg-gray-100 text-gray-400'
   162→              }`}
   163→            >
   164→              {step > s.num ? <Check className="w-4 h-4" /> : s.num}
   165→            </div>
   166→            <span className={`text-[11px] font-medium ${step >= s.num ? 'text-emerald-700' : 'text-gray-400'}`}>
   167→              {s.label}
   168→            </span>
   169→          </div>
   170→          {i < steps.length - 1 && (
   171→            <div className={`flex-1 h-0.5 mx-3 rounded-full transition-colors duration-300 ${step > s.num ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   172→          )}
   173→        </div>
   174→      ))}
   175→    </div>
   176→  )
   177→}

   608→                <TabsContent value="reviews" className="mt-3">
   609→                  {/* Reviews list */}

   611→                    <div className="space-y-3 max-h-64 overflow-y-auto">

   653→                  {/* Review form */}

   700→                          />

   715→          </div>
