   212→  name: string
   213→  description: string
   214→  lotSize: number
   215→  lotUnit: string
   216→  pricePerLot: number
   217→  comparePrice: number
   218→  minLots: number

   223→
   224→const emptyProductForm: ProductFormData = {
   225→  name: '', description: '', lotSize: 1, lotUnit: 'unités', pricePerLot: 0,
   226→  comparePrice: 0, minLots: 1, stockLots: 0, image: '', isActive: true,

   228→
   229→function WholesaleProductsTab() {
   230→  const queryClient = useQueryClient()
   231→  const [createOpen, setCreateOpen] = useState(false)
   232→  const [editOpen, setEditOpen] = useState(false)
   233→  const [deleteOpen, setDeleteOpen] = useState(false)
   234→  const [editingProduct, setEditingProduct] = useState<WholesaleProduct | null>(null)
   235→  const [deletingProduct, setDeletingProduct] = useState<WholesaleProduct | null>(null)
   236→  const [form, setForm] = useState<ProductFormData>(emptyProductForm)
   237→  const [formLoading, setFormLoading] = useState(false)
   238→  const [search, setSearch] = useState('')
   239→
   240→  const { data: products, isLoading } = useQuery({
   241→    queryKey: ['wholesale-products'],
   242→    queryFn: () => fetch('/api/wholesale/products').then(r => r.json()).then(d => d.data as WholesaleProduct[]),
   243→  })
   244→
   245→  const filteredProducts = useMemo(() => {
   246→    if (!products) return []
   247→    if (!search.trim()) return products
   248→    const q = search.toLowerCase()
   249→    return products.filter(p => p.name.toLowerCase().includes(q) || (p.description || '').toLowerCase().includes(q))
   250→  }, [products, search])
   251→
   252→  const totalActive = products?.filter(p => p.isActive).length ?? 0
   253→  const totalInactive = products?.filter(p => !p.isActive).length ?? 0
   254→
   255→  const resetForm = () => setForm(emptyProductForm)
   256→
   257→  const openEdit = (p: WholesaleProduct) => {
   258→    setForm({
   259→      name: p.name,
   260→      description: p.description || '',
   261→      lotSize: p.lotSize,
   262→      lotUnit: p.lotUnit,
   263→      pricePerLot: p.pricePerLot,
   264→      comparePrice: p.comparePrice || 0,
   265→      minLots: p.minLots,
   266→      stockLots: p.stockLots,
   267→      image: p.image || '',
   268→      isActive: p.isActive,
   269→    })
   270→    setEditingProduct(p)
   271→    setEditOpen(true)
   272→  }
   273→
   274→  const handleCreate = async () => {
   275→    if (!form.name.trim()) { toast.error('Le nom est obligatoire'); return }
   276→    if (form.pricePerLot <= 0) { toast.error('Le prix par lot doit être positif'); return }
   277→    setFormLoading(true)
   278→    try {
   279→      const res = await fetch('/api/wholesale/products', {
   280→        method: 'POST',
   281→        headers: { 'Content-Type': 'application/json' },
   282→        body: JSON.stringify({
   283→          name: form.name.trim(),
   284→          description: form.description.trim() || null,
   285→          lotSize: form.lotSize,
   286→          lotUnit: form.lotUnit,
   287→          pricePerLot: form.pricePerLot,
   288→          comparePrice: form.comparePrice > 0 ? form.comparePrice : null,
   289→          minLots: form.minLots,
   290→          stockLots: form.stockLots,
   291→          image: form.image.trim() || null,
   292→          isActive: form.isActive,
   293→        }),
   294→      })
   295→      const data = await res.json()
   296→      if (data.success) {
   297→        toast.success('Produit en gros ajouté')
   298→        queryClient.invalidateQueries({ queryKey: ['wholesale-products'] })
   299→        setCreateOpen(false)
   300→        resetForm()
   301→      } else {
   302→        toast.error(data.error || 'Erreur lors de l\'ajout')
   303→      }
   304→    } catch {
   305→      toast.error('Erreur serveur')
   306→    } finally {
   307→      setFormLoading(false)
   308→    }
   309→  }
   310→
   311→  const handleEdit = async () => {
   312→    if (!editingProduct || !form.name.trim()) { toast.error('Le nom est obligatoire'); return }
   313→    setFormLoading(true)
   314→    try {
   315→      const res = await fetch(`/api/wholesale/products/${editingProduct.id}`, {
   316→        method: 'PUT',
   317→        headers: { 'Content-Type': 'application/json' },
   318→        body: JSON.stringify({
   319→          name: form.name.trim(),
   320→          description: form.description.trim() || null,
   321→          lotSize: form.lotSize,
   322→          lotUnit: form.lotUnit,
   323→          pricePerLot: form.pricePerLot,
   324→          comparePrice: form.comparePrice > 0 ? form.comparePrice : null,
   325→          minLots: form.minLots,
   326→          stockLots: form.stockLots,
   327→          image: form.image.trim() || null,
   328→          isActive: form.isActive,
   329→        }),
   330→      })
   331→      const data = await res.json()
   332→      if (data.success) {
   333→        toast.success('Produit modifié')
   334→        queryClient.invalidateQueries({ queryKey: ['wholesale-products'] })
   335→        setEditOpen(false)
   336→        setEditingProduct(null)
   337→        resetForm()
   338→      } else {
   339→        toast.error(data.error || 'Erreur lors de la modification')
   340→      }
   341→    } catch {
   342→      toast.error('Erreur serveur')
   343→    } finally {
   344→      setFormLoading(false)
   345→    }
   346→  }
   347→
   348→  const handleToggle = async (p: WholesaleProduct) => {
   349→    try {
   350→      const res = await fetch(`/api/wholesale/products/${p.id}`, {
   351→        method: 'PUT',
   352→        headers: { 'Content-Type': 'application/json' },
   353→        body: JSON.stringify({ isActive: !p.isActive }),
   354→      })
   355→      const data = await res.json()
   356→      if (data.success) {
   357→        toast.success(p.isActive ? 'Produit désactivé' : 'Produit activé')
   358→        queryClient.invalidateQueries({ queryKey: ['wholesale-products'] })
   359→      } else {
   360→        toast.error(data.error || 'Erreur')
   361→      }
   362→    } catch {
   363→      toast.error('Erreur serveur')
   364→    }
   365→  }
   366→
   367→  const handleDelete = async () => {
   368→    if (!deletingProduct) return
   369→    try {
   370→      const res = await fetch(`/api/wholesale/products/${deletingProduct.id}`, { method: 'DELETE' })
   371→      const data = await res.json()
   372→      if (data.success) {
   373→        toast.success('Produit supprimé')
   374→        queryClient.invalidateQueries({ queryKey: ['wholesale-products'] })
   375→      } else {
   376→        toast.error(data.error || 'Erreur lors de la suppression')
   377→      }
   378→    } catch {
   379→      toast.error('Erreur serveur')
   380→    } finally {
   381→      setDeleteOpen(false)
   382→      setDeletingProduct(null)
   383→    }
   384→  }
   385→
   386→  return (
   387→    <div className="space-y-6">
   388→      {/* Stats */}
   389→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   390→        <Card>
   391→          <CardContent className="p-4">
   392→            <p className="text-sm text-muted-foreground">Total produits</p>
   393→            <p className="text-2xl font-bold text-gray-900">{products?.length ?? 0}</p>
   394→          </CardContent>
   395→        </Card>
   396→        <Card>
   397→          <CardContent className="p-4">
   398→            <p className="text-sm text-muted-foreground">Actifs</p>
   399→            <p className="text-2xl font-bold text-emerald-600">{totalActive}</p>
   400→          </CardContent>
   401→        </Card>
   402→        <Card>
   403→          <CardContent className="p-4">
   404→            <p className="text-sm text-muted-foreground">Inactifs</p>
   405→            <p className="text-2xl font-bold text-gray-500">{totalInactive}</p>
   406→          </CardContent>
   407→        </Card>
   408→        <Card>
   409→          <CardContent className="p-4">
   410→            <p className="text-sm text-muted-foreground">Stock total (lots)</p>
   411→            <p className="text-2xl font-bold text-gray-900">{products?.reduce((s, p) => s + p.stockLots, 0) ?? 0}</p>
   412→          </CardContent>
   413→        </Card>
   414→      </div>
   415→
   416→      {/* Header */}
   417→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   418→        <div>
   419→          <h3 className="text-lg font-semibold text-gray-900">Produits en Gros</h3>
   420→          <p className="text-sm text-muted-foreground">Gérer les produits vendus en lots</p>
   421→        </div>
   422→        <div className="flex-1" />
   423→        <div className="flex items-center gap-2 w-full sm:w-auto">
   424→          <div className="relative flex-1 sm:flex-initial">
   425→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   426→            <Input
   427→              placeholder="Rechercher..."
   428→              value={search}
   429→              onChange={e => setSearch(e.target.value)}
   430→              className="pl-9 w-full sm:w-[200px]"
   431→            />
   432→          </div>
   433→          <Button
   434→            onClick={() => { resetForm(); setCreateOpen(true) }}
   435→            className="bg-[#1a1a1a] hover:bg-[#1a1a1a]/90 text-white"
   436→          >
   437→            <Plus className="h-4 w-4 mr-2" />
   438→            Ajouter
   439→          </Button>
   440→        </div>
   441→      </div>
   442→
   443→      {/* Table */}
   444→      <Card>
   445→        <CardContent className="p-0">
   446→          {isLoading ? (
   447→            <div className="p-6 space-y-4">{[1, 2, 3].map(i => <Skeleton key={i} className="h-14 w-full" />)}</div>
   448→          ) : filteredProducts.length > 0 ? (
   449→            <div className="overflow-x-auto">
   450→              <Table>
   451→                <TableHeader>
   452→                  <TableRow className="bg-gray-50/80">
   453→                    <TableHead>Produit</TableHead>
   454→                    <TableHead>Lot</TableHead>
   455→                    <TableHead>Prix/lot</TableHead>
   456→                    <TableHead className="hidden md:table-cell">Min. lots</TableHead>
   457→                    <TableHead>Stock (lots)</TableHead>
   458→                    <TableHead>Statut</TableHead>
   459→                    <TableHead className="w-36">Actions</TableHead>
   460→                  </TableRow>
   461→                </TableHeader>
   462→                <TableBody>
   463→                  {filteredProducts.map((p, i) => (
   464→                    <TableRow key={p.id} className={`${i % 2 === 0 ? 'bg-white' : 'bg-gray-50/30'} hover:bg-gray-50`}>
   465→                      <TableCell>
   466→                        <div className="flex items-center gap-3">
   467→                          {p.image ? (
   468→                            <img src={p.image} alt={p.name} className="h-10 w-10 rounded-lg object-cover border border-gray-100 shrink-0" />
   469→                          ) : (
   470→                            <div className="h-10 w-10 rounded-lg bg-gray-100 flex items-center justify-center shrink-0">
   471→                              <Box className="h-5 w-5 text-gray-400" />
   472→                            </div>
   473→                          )}
   474→                          <div className="min-w-0">
   475→                            <p className="text-sm font-medium text-gray-900 truncate">{p.name}</p>
   476→                            {p.description && (
   477→                              <p className="text-xs text-muted-foreground truncate max-w-[200px]">{p.description}</p>
   478→                            )}
   479→                          </div>
   480→                        </div>
   481→                      </TableCell>
   482→                      <TableCell>
   483→                        <Badge variant="outline" className="bg-gray-50 text-gray-700 border-gray-200 whitespace-nowrap">
   484→                          <Layers className="h-3 w-3 mr-1" />
   485→                          1 lot = {p.lotSize} {p.lotUnit}
   486→                        </Badge>
   487→                      </TableCell>
   488→                      <TableCell>
   489→                        <div>
   490→                          <p className="text-sm font-semibold text-gray-900">{formatPrice(p.pricePerLot)}</p>
   491→                          {p.comparePrice && p.comparePrice > 0 && (
   492→                            <p className="text-xs text-muted-foreground line-through">{formatPrice(p.comparePrice)}</p>
   493→                          )}
   494→                        </div>
   495→                      </TableCell>
   496→                      <TableCell className="hidden md:table-cell">
   497→                        <span className="text-sm text-muted-foreground">{p.minLots}</span>
   498→                      </TableCell>
   499→                      <TableCell>
   500→                        <span className={`text-sm font-medium ${p.stockLots <= p.minLots ? 'text-red-600' : 'text-gray-900'}`}>
   501→                          {p.stockLots}
   502→                        </span>
   503→                      </TableCell>
   504→                      <TableCell>
   505→                        <Badge
   506→                          variant="outline"
   507→                          className={p.isActive
   508→                            ? 'bg-emerald-50 text-emerald-700 border-emerald-200'
   509→                            : 'bg-red-50 text-red-600 border-red-200'
   510→                          }
   511→                        >
   512→                          {p.isActive ? 'Actif' : 'Inactif'}
   513→                        </Badge>
   514→                      </TableCell>
   515→                      <TableCell>
   516→                        <div className="flex items-center gap-1">
   517→                          <Button
   518→                            variant="ghost"
   519→                            size="icon"
   520→                            className={`h-8 w-8 ${p.isActive ? 'text-gray-500 hover:text-orange-600' : 'text-orange-500 hover:text-orange-700'}`}
   521→                            onClick={() => handleToggle(p)}
   522→                            title={p.isActive ? 'Désactiver' : 'Activer'}
   523→                          >
   524→                            {p.isActive ? <PowerOff className="h-4 w-4" /> : <Power className="h-4 w-4" />}
   525→                          </Button>
   526→                          <Button
   527→                            variant="ghost"
   528→                            size="icon"

   531→                            title="Modifier"

  1572→      {/* Create Dialog */}
