   499→        {/* Livraison Tab */}

   526→                  </div>
   527→                </div>
   528→              </CardContent>
   529→            </Card>
   530→          </div>
   531→        </TabsContent>
   532→      </Tabs>
   533→
   534→      {/* Admin User Management */}
   535→      {isSuperAdmin && (

   539→              <div className="flex items-center justify-between">

   548→            </CardHeader>
   549→            <CardContent>

   584→                  <Users className="h-10 w-10 mx-auto text-gray-300 mb-2" />

   587→              )}
   588→            </CardContent>
   589→          </Card>
   590→        </div>
   591→      )}
   592→
   593→      {/* Add Admin Dialog */}
   594→      <Dialog open={adminDialogOpen} onOpenChange={setAdminDialogOpen}>

   616→              />
   617→            </div>

   619→              <Label>Mot de passe *</Label>
   620→              <Input
   621→                type="password"
   622→                value={newAdmin.password}
   623→                onChange={e => setNewAdmin(prev => ({ ...prev, password: e.target.value }))}

   625→              />
   626→            </div>
   627→            <div className="space-y-2">
   628→              <Label>Confirmer le mot de passe *</Label>
   629→              <Input

   635→            </div>
   636→          </div>

   640→              {creatingAdmin ? <RefreshCw className="h-4 w-4 animate-spin mr-2" /> : null}
   641→              Créer

   643→          </DialogFooter>
