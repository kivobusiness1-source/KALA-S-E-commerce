     1→'use client'
     2→
     3→import { useState, useEffect, useRef, useCallback } from 'react'
     4→import { useQuery, useQueryClient } from '@tanstack/react-query'
     5→import { format } from 'date-fns'
     6→import { fr } from 'date-fns/locale'
     7→import { toast } from 'sonner'
     8→import {
     9→  ShoppingCart, Search, Eye, ChevronLeft, ChevronRight, MoreVertical, FileDown, Printer,
    10→  MessageSquare, Send, Trash2, Clock, Shield, Bot, Reply, Paperclip, UserPlus,
    11→} from 'lucide-react'
    12→import { Button } from '@/components/ui/button'
    13→import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@/components/ui/card'
    14→import { Badge } from '@/components/ui/badge'
    15→import { Input } from '@/components/ui/input'
    16→import { Textarea } from '@/components/ui/textarea'
    17→import { Label } from '@/components/ui/label'
    18→import { Separator } from '@/components/ui/separator'
    19→import { Skeleton } from '@/components/ui/skeleton'
    20→import {
    21→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription,
    22→} from '@/components/ui/dialog'
    23→import {
    24→  DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuTrigger,
    25→} from '@/components/ui/dropdown-menu'
    26→import {
    27→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    28→} from '@/components/ui/select'
    29→import {
    30→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    31→} from '@/components/ui/table'
    32→import { formatPrice, formatDate, getInitials, statusLabels, StatusBadge } from './helpers'
    33→import { useAdminStore } from '@/stores/admin-store'
    34→import type { Order, OrderStatus, OrderNote, AdminUser } from './types'
    35→
    36→const MAX_REPLY_LENGTH = 2000
    37→
    38→export default function OrdersSection() {
    39→  const { admin } = useAdminStore()
    40→  const queryClient = useQueryClient()
    41→  const [statusFilter, setStatusFilter] = useState('all')
    42→  const [search, setSearch] = useState('')
    43→  const [page, setPage] = useState(1)
    44→  const [orderDetail, setOrderDetail] = useState<Order | null>(null)
    45→  const [detailOpen, setDetailOpen] = useState(false)
    46→  const [showNotes, setShowNotes] = useState(false)
    47→  const [orderNotes, setOrderNotes] = useState<OrderNote[]>([])
    48→  const [replyText, setReplyText] = useState('')
    49→  const [sendingReply, setSendingReply] = useState(false)
    50→  const [loadingNotes, setLoadingNotes] = useState(false)
    51→  const [assigneeId, setAssigneeId] = useState<string>('none')
    52→  const notesEndRef = useRef<HTMLDivElement>(null)
    53→  const limit = 20
    54→
    55→  const role = admin?.role || ''
    56→  const isLivreur = role === 'livreur'
    57→  const isStaff = role === 'staff'
    58→  const isSuperAdmin = role === 'super_admin'
    59→  const canAssign = isSuperAdmin || role === 'admin'
    60→
    61→  // Staff cannot set delivered
    62→  const allowedStatuses = isStaff
    63→    ? ['pending', 'confirmed', 'processing', 'shipped', 'cancelled'] as OrderStatus[]
    64→    : ['pending', 'confirmed', 'processing', 'shipped', 'delivered', 'cancelled'] as OrderStatus[]
    65→
    66→  const { data: ordersData, isLoading } = useQuery({
    67→    queryKey: ['admin-orders', statusFilter, search, page],
    68→    queryFn: () => {
    69→      const params = new URLSearchParams({ page: String(page), limit: String(limit) })
    70→      if (statusFilter !== 'all') params.set('status', statusFilter)
    71→      if (search) params.set('search', search)
    72→      return fetch(`/api/orders?${params}`).then(r => r.json()).then(d => d.data)
    73→    },
    74→  })
    75→
    76→  // Fetch admins/livreurs for assignment dropdown
    77→  const { data: adminUsers } = useQuery({
    78→    queryKey: ['admin-users-for-assignment'],
    79→    queryFn: () => fetch('/api/admin/users').then(r => r.json()).then(d => d.data as AdminUser[]),
    80→    enabled: canAssign,
    81→  })
    82→
    83→  const orders = ordersData?.orders ?? []
    84→  const totalPages = ordersData?.totalPages ?? 1
    85→
    86→  const fetchNotes = useCallback(async (orderId: string) => {
    87→    setLoadingNotes(true)
    88→    try {
    89→      const res = await fetch(`/api/orders/${orderId}/notes`)
    90→      const data = await res.json()
    91→      if (data.success) {
    92→        setOrderNotes(data.data)
    93→      }
    94→    } catch {
    95→      // silently fail
    96→    } finally {
    97→      setLoadingNotes(false)
    98→    }
    99→  }, [])
   100→
   101→  const openDetail = async (orderId: string) => {
   102→    try {
   103→      const res = await fetch(`/api/orders/${orderId}`)
   104→      const data = await res.json()
   105→      if (data.success) {
   106→        setOrderDetail(data.data)
   107→        setAssigneeId(data.data.assignedToId || 'none')
   108→        setDetailOpen(true)
   109→        setShowNotes(false)
   110→        setReplyText('')
   111→        fetchNotes(orderId)
   112→      }
   113→    } catch {
   114→      toast.error('Erreur de chargement')
   115→    }
   116→  }
   117→
   118→  // Scroll to bottom when notes change
   119→  useEffect(() => {
   120→    if (showNotes && notesEndRef.current) {
   121→      notesEndRef.current.scrollIntoView({ behavior: 'smooth' })
   122→    }
   123→  }, [orderNotes, showNotes])
   124→
   125→  const updateStatus = async (orderId: string, newStatus: string) => {
   126→    // Staff cannot deliver
   127→    if (isStaff && newStatus === 'delivered') {
   128→      toast.error('Le staff ne peut pas marquer comme livré')
   129→      return
   130→    }
   131→    const oldStatus = orderDetail?.status
   132→    try {
   133→      const res = await fetch(`/api/orders/${orderId}`, {
   134→        method: 'PUT',
   135→        headers: { 'Content-Type': 'application/json' },
   136→        body: JSON.stringify({ status: newStatus }),
   137→      })
   138→      const data = await res.json()
   139→      if (data.success) {
   140→        toast.success('Statut mis à jour')
   141→        queryClient.invalidateQueries({ queryKey: ['admin-orders'] })
   142→        queryClient.invalidateQueries({ queryKey: ['stats'] })
   143→        if (orderDetail?.id === orderId) {
   144→          setOrderDetail({ ...orderDetail, status: newStatus })
   145→        }
   146→
   147→        // Auto-add system note when status changes
   148→        if (oldStatus && oldStatus !== newStatus && orderId) {
   149→          const oldLabel = statusLabels[oldStatus] || oldStatus
   150→          const newLabel = statusLabels[newStatus] || newStatus
   151→          try {
   152→            await fetch(`/api/orders/${orderId}/notes`, {
   153→              method: 'POST',
   154→              headers: { 'Content-Type': 'application/json' },
   155→              body: JSON.stringify({ content: `Statut mis à jour : ${oldLabel} → ${newLabel}` }),
   156→            })
   157→            fetchNotes(orderId)
   158→          } catch {
   159→            // silent
   160→          }
   161→        }
   162→      } else {
   163→        toast.error(data.error || 'Erreur')
   164→      }
   165→    } catch {
   166→      toast.error('Erreur serveur')
   167→    }
   168→  }
   169→
   170→  const handleAssign = async () => {
   171→    if (!orderDetail) return
   172→    try {
   173→      const res = await fetch(`/api/orders/${orderDetail.id}`, {
   174→        method: 'PUT',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ assignedToId: assigneeId === 'none' ? null : assigneeId }),
   177→      })
   178→      const data = await res.json()
   179→      if (data.success) {
   180→        toast.success('Commande réassignée')
   181→        setOrderDetail(data.data)
   182→        queryClient.invalidateQueries({ queryKey: ['admin-orders'] })
   183→      } else {
   184→        toast.error(data.error || 'Erreur')
   185→      }
   186→    } catch {
   187→      toast.error('Erreur serveur')
   188→    }
   189→  }
   190→
   191→  const sendReply = async () => {
   192→    if (!orderDetail || !replyText.trim()) return
   193→    const content = replyText.trim()
   194→    if (content.length > MAX_REPLY_LENGTH) {
   195→      toast.error('Message trop long')
   196→      return
   197→    }
   198→    setSendingReply(true)
   199→    try {
   200→      const res = await fetch(`/api/orders/${orderDetail.id}/notes`, {
   201→        method: 'POST',
   202→        headers: { 'Content-Type': 'application/json' },
   203→        body: JSON.stringify({ content }),
   204→      })
   205→      const data = await res.json()
   206→      if (data.success) {
   207→        setReplyText('')
   208→        fetchNotes(orderDetail.id)
   209→        toast.success('Réponse envoyée')
   210→      } else {
   211→        toast.error(data.error || 'Erreur')
   212→      }
   213→    } catch {
   214→      toast.error('Erreur serveur')
   215→    } finally {
   216→      setSendingReply(false)
   217→    }
   218→  }
   219→
   220→  const deleteNote = async (noteId: string) => {
   221→    if (!orderDetail) return
   222→    try {
   223→      const res = await fetch(`/api/orders/${orderDetail.id}/notes/${noteId}`, { method: 'DELETE' })
   224→      if (res.ok) {
   225→        setOrderNotes(prev => prev.filter(n => n.id !== noteId))
   226→        toast.success('Note supprimée')
   227→      }
   228→    } catch {
   229→      toast.error('Erreur')
   230→    }
   231→  }
   232→
   233→  const statusTabs = ['all', 'pending', 'confirmed', 'processing', 'shipped', 'delivered', 'cancelled'] as const
   234→
   235→  const exportOrdersCSV = async () => {
   236→    try {
   237→      const res = await fetch('/api/orders?limit=1000')
   238→      const data = await res.json()
   239→      const allOrders: Order[] = data.data?.orders ?? []
   240→      if (allOrders.length === 0) { toast.error('Aucune commande à exporter'); return }
   241→      const headers = ['N°', 'Client', 'Email', 'Téléphone', 'Adresse', 'Montant', 'Statut', 'Date']
   242→      const rows = allOrders.map(o => [
   243→        o.orderNumber,
   244→        o.customerName,
   245→        o.customerEmail,
   246→        o.customerPhone || '',
   247→        `${o.address}, ${o.city}`,
   248→        String(o.totalAmount),
   249→        statusLabels[o.status] || o.status,
   250→        format(new Date(o.createdAt), 'yyyy-MM-dd HH:mm'),
   251→      ])
   252→      const csv = [headers, ...rows].map(r => r.map(c => `"${c}"`).join(',')).join('\n')
   253→      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   254→      const url = URL.createObjectURL(blob)
   255→      const a = document.createElement('a')
   256→      a.href = url
   257→      a.download = `commandes-congoclean-${format(new Date(), 'yyyy-MM-dd')}.csv`
   258→      a.click()
   259→      URL.revokeObjectURL(url)
   260→      toast.success('Fichier CSV téléchargé')
   261→    } catch {
   262→      toast.error('Erreur lors de l\'export')
   263→    }
   264→  }
   265→
   266→  const formatNoteTime = (dateStr: string) => {
   267→    try {
   268→      return format(new Date(dateStr), "d MMM yyyy 'à' HH:mm", { locale: fr })
   269→    } catch {
   270→      return dateStr
   271→    }
   272→  }
   273→
   274→  return (
   275→    <div className="space-y-4">
   276→      {/* Filters */}
   277→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   278→        <div className="flex items-center gap-2 w-full sm:w-auto overflow-x-auto pb-1 sm:pb-0">
   279→          {statusTabs.map(s => (
   280→            <Button key={s} variant={statusFilter === s ? 'default' : 'outline'} size="sm" className={`shrink-0 ${statusFilter === s ? 'bg-[#1a1a1a] hover:bg-[#1a1a1a]/90 text-white' : ''}`} onClick={() => { setStatusFilter(s); setPage(1) }}>
   281→              {s === 'all' ? 'Tous' : statusLabels[s]}
   282→            </Button>
   283→          ))}
   284→        </div>
   285→        <div className="flex-1" />
   286→        <div className="flex items-center gap-2 w-full sm:w-auto">
   287→          <div className="relative flex-1 sm:w-64">
   288→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   289→            <Input placeholder="Rechercher..." value={search} onChange={e => { setSearch(e.target.value); setPage(1) }} className="pl-9" />
   290→          </div>
   291→          <Button variant="outline" size="sm" onClick={exportOrdersCSV}>
   292→            <FileDown className="h-4 w-4 mr-1" />CSV
   293→          </Button>
   294→        </div>
   295→      </div>
   296→
   297→      {/* Livreur info banner */}
   298→      {isLivreur && (
   299→        <div className="bg-gray-100 rounded-lg p-3 text-sm text-gray-600">
   300→          Vous ne voyez que les commandes qui vous sont assignées.
   301→        </div>
   302→      )}
   303→
   304→      {/* Table */}
   305→      <Card>
   306→        <CardContent className="p-0">
   307→          {isLoading ? (
   308→            <div className="p-6 space-y-4">{[1, 2, 3, 4, 5].map(i => <Skeleton key={i} className="h-14 w-full" />)}</div>
   309→          ) : orders.length > 0 ? (
   310→            <div className="overflow-x-auto">
   311→              <Table>
   312→                <TableHeader>
   313→                  <TableRow>
   314→                    <TableHead>N° Commande</TableHead>
   315→                    <TableHead>Client</TableHead>
   316→                    <TableHead className="hidden md:table-cell">Email</TableHead>
   317→                    <TableHead className="hidden sm:table-cell">Date</TableHead>
   318→                    <TableHead className="text-right">Montant</TableHead>
   319→                    <TableHead>Articles</TableHead>
   320→                    <TableHead>Statut</TableHead>
   321→                    {canAssign && <TableHead className="hidden lg:table-cell">Assigné à</TableHead>}
   322→                    <TableHead className="w-28">Actions</TableHead>
   323→                  </TableRow>
   324→                </TableHeader>
   325→                <TableBody>
   326→                  {orders.map((o: Order & { assignedBy?: { name: string; role: string } }, i: number) => {
   327→                    return (
   328→                    <TableRow key={o.id} className={`${i % 2 === 0 ? 'bg-white' : 'bg-gray-50/30'} hover:bg-gray-50 transition-colors duration-150`}>
   329→                      <TableCell className="font-mono text-xs font-medium">{o.orderNumber}</TableCell>
   330→                      <TableCell className="text-sm font-medium">
   331→                        <div className="flex items-center gap-2">
   332→                          <div className="w-7 h-7 rounded-full bg-gray-100 text-gray-700 flex items-center justify-center text-xs font-semibold shrink-0">
   333→                            {getInitials(o.customerName)}
   334→                          </div>
   335→                          {o.customerName}
   336→                        </div>
   337→                      </TableCell>
   338→                      <TableCell className="hidden md:table-cell text-sm text-muted-foreground">{o.customerEmail}</TableCell>
   339→                      <TableCell className="hidden sm:table-cell text-sm text-muted-foreground">{formatDate(o.createdAt)}</TableCell>
   340→                      <TableCell className="text-right text-sm font-medium">{formatPrice(o.totalAmount)}</TableCell>
   341→                      <TableCell>
   342→                        <Badge variant="outline" className="bg-gray-50 text-gray-600 border-gray-200">
   343→                          {o.items?.length ?? 0} art.
   344→                        </Badge>
   345→                      </TableCell>
   346→                      <TableCell>
   347→                        <StatusBadge status={o.status} />
   348→                      </TableCell>
   349→                      {canAssign && (
   350→                        <TableCell className="hidden lg:table-cell text-sm text-muted-foreground">
   351→                          {(o as unknown as { assignedBy?: { name: string } }).assignedBy?.name || '—'}
   352→                        </TableCell>
   353→                      )}
   354→                      <TableCell>
   355→                        <div className="flex items-center gap-1">
   356→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openDetail(o.id)}><Eye className="h-4 w-4" /></Button>
   357→                          <DropdownMenu>
   358→                            <DropdownMenuTrigger asChild><Button variant="ghost" size="icon" className="h-8 w-8"><MoreVertical className="h-4 w-4" /></Button></DropdownMenuTrigger>
   359→                            <DropdownMenuContent align="end">
   360→                              {allowedStatuses.map(s => (
   361→                                <DropdownMenuItem key={s} onClick={() => updateStatus(o.id, s)} disabled={o.status === s}>
   362→                                  {statusLabels[s]}
   363→                                </DropdownMenuItem>
   364→                              ))}
   365→                            </DropdownMenuContent>
   366→                          </DropdownMenu>
   367→                        </div>
   368→                      </TableCell>
   369→                    </TableRow>
   370→                    )
   371→                  })}
   372→                </TableBody>
   373→              </Table>
   374→            </div>
   375→          ) : (
   376→            <div className="p-12 text-center">
   377→              <ShoppingCart className="h-12 w-12 mx-auto text-gray-300 mb-3" />
   378→              <p className="text-muted-foreground">Aucune commande trouvée</p>
   379→            </div>
   380→          )}
   381→        </CardContent>
   382→      </Card>
   383→
   384→      {/* Pagination */}
   385→      {totalPages > 1 && (
   386→        <div className="flex items-center justify-center gap-2">
   387→          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(p => p - 1)}>
   388→            <ChevronLeft className="h-4 w-4" />
   389→          </Button>
   390→          <span className="text-sm text-muted-foreground">Page {page} / {totalPages}</span>
   391→          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(p => p + 1)}>
   392→            <ChevronRight className="h-4 w-4" />
   393→          </Button>
   394→        </div>
   395→      )}
   396→
   397→      {/* Order Detail Dialog */}
   398→      <Dialog open={detailOpen} onOpenChange={(v) => { setDetailOpen(v); if (!v) { setShowNotes(false); setOrderNotes([]) } }}>
   399→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto" data-printable>
   400→          {orderDetail && (
   401→            <>
   402→              <DialogHeader className="pb-4 border-b border-gray-100">
   403→                <div className="absolute inset-x-0 top-0 h-1 bg-[#1a1a1a] rounded-t-xl" />
   404→                <div className="flex items-center justify-between pr-6">
   405→                  <div>
   406→                    <DialogTitle className="flex items-center gap-2">
   407→                      Commande {orderDetail.orderNumber}
   408→                      {orderNotes.length > 0 && (
   409→                        <Badge variant="outline" className="bg-gray-100 text-gray-700 border-gray-200 text-xs gap-1">
   410→                          <MessageSquare className="h-3 w-3" />
   411→                          {orderNotes.length}
   412→                        </Badge>
   413→                      )}
   414→                    </DialogTitle>
   415→                    <DialogDescription>Créée le {formatDate(orderDetail.createdAt)}</DialogDescription>
   416→                  </div>
   417→                  <div className="flex items-center gap-2">
   418→                    <Button
   419→                      variant={showNotes ? 'default' : 'outline'}
   420→                      size="sm"
   421→                      className={showNotes ? 'bg-[#1a1a1a] hover:bg-[#1a1a1a]/90 text-white gap-1.5' : 'gap-1.5'}
   422→                      onClick={() => setShowNotes(!showNotes)}
   423→                    >
   424→                      <Reply className="w-4 h-4" />
   425→                      Répondre
   426→                      {orderNotes.length > 0 && (
   427→                        <span className={`ml-1 text-xs px-1.5 py-0.5 rounded-full ${showNotes ? 'bg-white/20 text-white' : 'bg-gray-100 text-gray-700'}`}>
   428→                          {orderNotes.length}
   429→                        </span>
   430→                      )}
   431→                    </Button>
   432→                    <Button variant="outline" size="sm" className="gap-2" onClick={() => window.print()}>
   433→                      <Printer className="w-4 h-4" />
   434→                      <span className="hidden sm:inline">Imprimer</span>
   435→                    </Button>
   436→                  </div>
   437→                </div>
   438→              </DialogHeader>
   439→
   440→              <div className="space-y-4">
   441→                {/* Customer info */}
   442→                <Card>
   443→                  <CardHeader className="pb-3"><CardTitle className="text-sm">Informations Client</CardTitle></CardHeader>
   444→                  <CardContent className="text-sm space-y-1">
   445→                    <p><span className="text-muted-foreground">Nom:</span> <span className="font-medium">{orderDetail.customerName}</span></p>
   446→                    <p><span className="text-muted-foreground">Email:</span> {orderDetail.customerEmail}</p>
   447→                    {orderDetail.customerPhone && <p><span className="text-muted-foreground">Téléphone:</span> {orderDetail.customerPhone}</p>}
   448→                    <p><span className="text-muted-foreground">Adresse:</span> {orderDetail.address}, {orderDetail.city}</p>
   449→                  </CardContent>
   450→                </Card>
   451→
   452→                {/* Status + Assignment */}
   453→                <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   454→                  <div className="flex items-center gap-2">
   455→                    <span className="text-sm text-muted-foreground">Statut:</span>
   456→                    <StatusBadge status={orderDetail.status} />
   457→                  </div>
   458→                  <Select value={orderDetail.status} onValueChange={v => updateStatus(orderDetail.id, v)}>
   459→                    <SelectTrigger className="w-[180px]"><SelectValue /></SelectTrigger>
   460→                    <SelectContent>
   461→                      {allowedStatuses.map(s => (
   462→                        <SelectItem key={s} value={s}>{statusLabels[s]}</SelectItem>
   463→                      ))}
   464→                    </SelectContent>
   465→                  </Select>
   466→                </div>
   467→
   468→                {/* Assignment (admin/super_admin only) */}
   469→                {canAssign && (
   470→                  <Card>
   471→                    <CardHeader className="pb-3">
   472→                      <CardTitle className="text-sm flex items-center gap-2">
   473→                        <UserPlus className="h-4 w-4" />
   474→                        Assigner à
   475→                      </CardTitle>
   476→                    </CardHeader>
   477→                    <CardContent>
   478→                      <div className="flex items-center gap-2">
   479→                        <Select value={assigneeId} onValueChange={setAssigneeId}>
   480→                          <SelectTrigger className="flex-1"><SelectValue placeholder="Non assigné" /></SelectTrigger>
   481→                          <SelectContent>
   482→                            <SelectItem value="none">— Non assigné —</SelectItem>
   483→                            {adminUsers?.filter(u => u.isActive).map(u => (
   484→                              <SelectItem key={u.id} value={u.id}>{u.name} ({u.role})</SelectItem>
   485→                            ))}
   486→                          </SelectContent>
   487→                        </Select>
   488→                        <Button size="sm" onClick={handleAssign} className="bg-[#1a1a1a] hover:bg-[#1a1a1a]/90 text-white">Assigner</Button>
   489→                      </div>
   490→                    </CardContent>
   491→                  </Card>
   492→                )}
   493→
   494→                {/* Items */}
   495→                <Card>
   496→                  <CardHeader className="pb-3"><CardTitle className="text-sm">Articles</CardTitle></CardHeader>
   497→                  <CardContent>
   498→                    <Table>
   499→                      <TableHeader>
   500→                        <TableRow>
