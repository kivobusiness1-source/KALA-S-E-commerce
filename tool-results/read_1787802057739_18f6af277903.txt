   176→function isNewProduct(createdAt: string | null | undefined): boolean {
   177→  if (!createdAt) return false
   178→  const created = new Date(createdAt)
   179→  const thirtyDaysAgo = new Date()
   180→  thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30)
   181→  return created >= thirtyDaysAgo
   182→}

   195→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   196→  const [count, setCount] = useState(0)
   197→  const ref = useRef(null)
   198→  const isInView = useInView(ref, { once: true })
   199→
   200→  useEffect(() => {
   201→    if (!isInView) return
   202→    let start = 0
   203→    const duration = 2000
   204→    const step = target / (duration / 16)
   205→    const timer = setInterval(() => {
   206→      start += step
   207→      if (start >= target) {
   208→        setCount(target)
   209→        clearInterval(timer)
   210→      } else {
   211→        setCount(Math.floor(start))
   212→      }
   213→    }, 16)
   214→    return () => clearInterval(timer)
   215→  }, [isInView, target])
   216→
   217→  return <span ref={ref}>{count}{suffix}</span>
   218→}

   222→function TypingEffect({ phrases }: { phrases: string[] }) {
   223→  const [currentPhraseIndex, setCurrentPhraseIndex] = useState(0)
   224→  const [currentText, setCurrentText] = useState('')
   225→  const [isDeleting, setIsDeleting] = useState(false)
   226→
   227→  useEffect(() => {
   228→    const phrase = phrases[currentPhraseIndex]
   229→    if (!isDeleting) {
   230→      if (currentText.length < phrase.length) {
   231→        const timeout = setTimeout(() => {
   232→          setCurrentText(phrase.slice(0, currentText.length + 1))
   233→        }, 50)
   234→        return () => clearTimeout(timeout)
   235→      } else {
   236→        const timeout = setTimeout(() => {
   237→          setIsDeleting(true)
   238→        }, 2000)
   239→        return () => clearTimeout(timeout)
   240→      }
   241→    } else {
   242→      if (currentText.length > 0) {
   243→        const timeout = setTimeout(() => {
   244→          setCurrentText(currentText.slice(0, -1))
   245→        }, 50)
   246→        return () => clearTimeout(timeout)
   247→      } else {
   248→        setIsDeleting(false)
   249→        setCurrentPhraseIndex((prev) => (prev + 1) % phrases.length)
   250→      }
   251→    }
   252→  }, [currentText, isDeleting, currentPhraseIndex, phrases])
   253→
   254→  return (
   255→    <span className="text-emerald-300 font-bold text-xl">
   256→      {currentText}
   257→      <span className="animate-pulse">|</span>
   258→    </span>
   259→  )
   260→}

   264→function FadeInSection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   265→  const ref = useRef(null)
   266→  const isInView = useInView(ref, { once: true, margin: '-80px' })
   267→
   268→  return (
   269→    <motion.div
   270→      ref={ref}
   271→      initial={{ opacity: 0, y: 40 }}
   272→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   273→      transition={{ duration: 0.6, ease: 'easeOut' }}
   274→      className={className}
   275→    >
   276→      {children}
   277→    </motion.div>
   278→  )
   279→}
