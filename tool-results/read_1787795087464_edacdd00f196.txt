   130→// ==================== ANIMATED COUNTER ====================
   131→
   132→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   133→  const [count, setCount] = useState(0)
   134→  const ref = useRef(null)
   135→  const isInView = useInView(ref, { once: true })
   136→
   137→  useEffect(() => {
   138→    if (!isInView) return
   139→    let start = 0
   140→    const duration = 2000
   141→    const step = target / (duration / 16)
   142→    const timer = setInterval(() => {
   143→      start += step
   144→      if (start >= target) {
   145→        setCount(target)
   146→        clearInterval(timer)
   147→      } else {
   148→        setCount(Math.floor(start))
   149→      }
   150→    }, 16)
   151→    return () => clearInterval(timer)
   152→  }, [isInView, target])
   153→
   154→  return <span ref={ref}>{count}{suffix}</span>
   155→}

   159→function FadeInSection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   160→  const ref = useRef(null)
   161→  const isInView = useInView(ref, { once: true, margin: '-80px' })
   162→
   163→  return (
   164→    <motion.div
   165→      ref={ref}
   166→      initial={{ opacity: 0, y: 40 }}
   167→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   168→      transition={{ duration: 0.6, ease: 'easeOut' }}
   169→      className={className}
   170→    >
   171→      {children}
   172→    </motion.div>
   173→  )
   174→}
