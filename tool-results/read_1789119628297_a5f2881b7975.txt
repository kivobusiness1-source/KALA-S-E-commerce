   116→function formatFCFA(amount: number): string {
   117→  return new Intl.NumberFormat('fr-FR').format(Math.round(amount)) + ' FCFA'
   118→}
   119→
   120→function formatDate(dateStr: string): string {
   121→  const d = new Date(dateStr)
   122→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: 'short', year: 'numeric' })
   123→}
   124→
   125→function getPeriodDates(period: PeriodFilter): { from?: Date; to: Date } {
   126→  const now = new Date()
   127→  const to = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59)
   128→  switch (period) {
   129→    case 'today': {
   130→      const from = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0)
   131→      return { from, to }
   132→    }
   133→    case 'week': {
   134→      const dayOfWeek = now.getDay()
   135→      const diff = dayOfWeek === 0 ? 6 : dayOfWeek - 1
   136→      const from = new Date(now.getFullYear(), now.getMonth(), now.getDate() - diff, 0, 0, 0)
   137→      return { from, to }
   138→    }
   139→    case 'month': {
   140→      const from = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0)
   141→      return { from, to }
   142→    }
   143→    default:
   144→      return { to }
   145→  }
   146→}
