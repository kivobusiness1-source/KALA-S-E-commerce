   124→function FadeInSection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   125→  const ref = useRef(null)
   126→  const isInView = useInView(ref, { once: true, margin: '-80px' })
   127→
   128→  return (
   129→    <motion.div
   130→      ref={ref}
   131→      initial={{ opacity: 0, y: 40 }}
   132→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   133→      transition={{ duration: 0.6, ease: 'easeOut' }}
   134→      className={className}
   135→    >
   136→      {children}
   137→    </motion.div>
   138→  )
   139→}
