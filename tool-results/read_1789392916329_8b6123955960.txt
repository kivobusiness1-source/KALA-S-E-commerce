    34→function parseImages(imagesStr: string | null | undefined): string[] {
    35→  if (!imagesStr) return []
    36→  try {
    37→    const parsed = JSON.parse(imagesStr)
    38→    return Array.isArray(parsed) ? parsed.filter((u: unknown): u is string => typeof u === 'string') : []
    39→  } catch {
    40→    return []
    41→  }
    42→}
