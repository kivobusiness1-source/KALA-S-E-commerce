   146→   146→function OrderProgressIndicator({ step }: { step: 1 | 2 | 3 }) {
   147→   147→  const steps = [
   148→   148→    { num: 1, label: 'Informations' },
   149→   149→    { num: 2, label: 'Livraison' },
   150→   150→    { num: 3, label: 'Confirmation' },

   152→   152→  return (
   153→   153→    <div className="flex items-center justify-between mb-6">
   154→   154→      {steps.map((s, i) => (
   155→   155→        <div key={s.num} className="flex items-center flex-1 last:flex-none">
   156→   156→          <div className="flex flex-col items-center gap-1.5">
   157→   157→            <div
   158→   158→              className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-semibold transition-colors duration-300 ${
   159→   159→                step >= s.num
   160→   160→                  ? 'bg-emerald-600 text-white shadow-md shadow-emerald-600/25'
   161→   161→                  : 'bg-gray-100 text-gray-400'
   162→   162→              }`}
   163→   163→            >
   164→   164→              {step > s.num ? <Check className="w-4 h-4" /> : s.num}
   165→   165→            </div>
   166→   166→            <span className={`text-[11px] font-medium ${step >= s.num ? 'text-emerald-700' : 'text-gray-400'}`}>
   167→   167→              {s.label}
   168→   168→            </span>
   169→   169→          </div>
   170→   170→          {i < steps.length - 1 && (
   171→   171→            <div className={`flex-1 h-0.5 mx-3 rounded-full transition-colors duration-300 ${step > s.num ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   172→   172→          )}
   173→   173→        </div>
   174→   174→      ))}
   175→   175→    </div>
   176→   176→  )
   177→   177→}

   419→   419→            </DialogDescription>
