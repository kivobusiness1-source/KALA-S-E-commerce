   126→   126→function formatDate(dateStr: string): string {
   127→   127→  const d = new Date(dateStr)
   128→   128→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: 'short', year: 'numeric' })
   129→   129→}
   130→   130→
   131→   131→function getPeriodDates(period: PeriodFilter): { from?: Date; to: Date } {
   132→   132→  const now = new Date()
   133→   133→  const to = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59)
   134→   134→  switch (period) {
   135→   135→    case 'today': {
   136→   136→      const from = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0)
   137→   137→      return { from, to }
   138→   138→    }
   139→   139→    case 'week': {
   140→   140→      const dayOfWeek = now.getDay()
   141→   141→      const diff = dayOfWeek === 0 ? 6 : dayOfWeek - 1
   142→   142→      const from = new Date(now.getFullYear(), now.getMonth(), now.getDate() - diff, 0, 0, 0)
   143→   143→      return { from, to }
   144→   144→    }
   145→   145→    case 'month': {
   146→   146→      const from = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0)
   147→   147→      return { from, to }
   148→   148→    }
   149→   149→    default:
   150→   150→      return { to }
   151→   151→  }
   152→   152→}
