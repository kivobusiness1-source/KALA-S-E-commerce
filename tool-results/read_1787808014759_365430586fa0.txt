    39→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
    40→  const [deletingProduct, setDeletingProduct] = useState<Product | null>(null)
    41→  const [categoryDialogOpen, setCategoryDialogOpen] = useState(false)
    42→  const [editingCategory, setEditingCategory] = useState<Category | null>(null)
    43→  const [newCategoryName, setNewCategoryName] = useState('')

    45→  const [deleteCategoryDialogOpen, setDeleteCategoryDialogOpen] = useState(false)
    46→  const [deletingCategory, setDeletingCategory] = useState<Category | null>(null)
    47→  const [showCategoryManager, setShowCategoryManager] = useState(false)
    48→  const [stockAdjustProduct, setStockAdjustProduct] = useState<Product | null>(null)

    53→  // Product form state

    56→    volume: '', stockQty: '', minStockAlert: '', isFeatured: false, isActive: true,

    63→    queryFn: () => {

    65→      if (search) params.set('search', search)
    66→      if (categoryFilter && categoryFilter !== 'all') params.set('categoryId', categoryFilter)
    67→      return fetch(`/api/products?${params}`).then(r => r.json()).then(d => d.data as Product[])
    68→    },
    69→  })
    70→

    72→    queryKey: ['categories'],
    73→    queryFn: () => fetch('/api/categories').then(r => r.json()).then(d => d.data as Category[]),

    78→      setEditingProduct(product)
    79→      setPForm({
    80→        name: product.name,
    81→        description: product.description || '',
    82→        longDescription: product.longDescription || '',
    83→        price: String(product.price),
    84→        comparePrice: product.comparePrice ? String(product.comparePrice) : '',
    85→        categoryId: product.categoryId,
    86→        volume: product.volume || '',
    87→        stockQty: String(product.stockQty),
    88→        minStockAlert: String(product.minStockAlert),
    89→        isFeatured: product.isFeatured,
    90→        isActive: product.isActive,
    91→      })
    92→    } else {
    93→      setEditingProduct(null)
    94→      setPForm({ name: '', description: '', longDescription: '', price: '', comparePrice: '', categoryId: '', volume: '', stockQty: '0', minStockAlert: '10', isFeatured: false, isActive: true })
    95→    }
    96→    setProductDialogOpen(true)
    97→  }
    98→
    99→  const saveProduct = async () => {
   100→    if (!pForm.name || !pForm.price || !pForm.categoryId) {
   101→      toast.error('Nom, prix et catégorie sont requis')
   102→      return
   103→    }
   104→    setSaving(true)
   105→    try {
   106→      const body = {
   107→        name: pForm.name,
   108→        description: pForm.description || undefined,
   109→        longDescription: pForm.longDescription || undefined,
   110→        price: parseFloat(pForm.price),
   111→        comparePrice: pForm.comparePrice ? parseFloat(pForm.comparePrice) : null,
   112→        categoryId: pForm.categoryId,
   113→        volume: pForm.volume || null,
   114→        stockQty: parseInt(pForm.stockQty) || 0,
   115→        minStockAlert: parseInt(pForm.minStockAlert) || 10,
   116→        isFeatured: pForm.isFeatured,
   117→        isActive: pForm.isActive,
   118→        image: editingProduct?.image || null,
   119→      }
   120→
   121→      let res: Response
   122→      if (editingProduct) {
   123→        res = await fetch(`/api/products/${editingProduct.id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   124→      } else {
   125→        res = await fetch('/api/products', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   126→      }
   127→      const data = await res.json()
   128→      if (data.success) {
   129→        toast.success(editingProduct ? 'Produit modifié' : 'Produit créé')
   130→        setProductDialogOpen(false)
   131→        queryClient.invalidateQueries({ queryKey: ['admin-products'] })
   132→        queryClient.invalidateQueries({ queryKey: ['stats'] })
   133→      } else {
   134→        toast.error(data.error || 'Erreur')
   135→      }
   136→    } catch {
   137→      toast.error('Erreur serveur')
   138→    } finally {
   139→      setSaving(false)
   140→    }
   141→  }
   142→
   143→  const deleteProduct = async () => {
   144→    if (!deletingProduct) return
   145→    try {
   146→      const res = await fetch(`/api/products/${deletingProduct.id}`, { method: 'DELETE' })
   147→      const data = await res.json()
   148→      if (data.success) {
   149→        toast.success('Produit supprimé')
   150→        queryClient.invalidateQueries({ queryKey: ['admin-products'] })
   151→        queryClient.invalidateQueries({ queryKey: ['stats'] })
   152→      } else {
   153→        toast.error(data.error || 'Erreur')
   154→      }
   155→    } catch {
   156→      toast.error('Erreur serveur')
   157→    } finally {
   158→      setDeleteDialogOpen(false)
   159→      setDeletingProduct(null)
   160→    }
   161→  }
   162→
   163→  const toggleProductActive = async (product: Product) => {
   164→    try {
   165→      const res = await fetch(`/api/products/${product.id}`, {
   166→        method: 'PUT',
   167→        headers: { 'Content-Type': 'application/json' },
   168→        body: JSON.stringify({ isActive: !product.isActive, inStock: !product.isActive }),
   169→      })
   170→      const data = await res.json()
   171→      if (data.success) {
   172→        toast.success(product.isActive ? 'Produit désactivé' : 'Produit activé')
   173→        queryClient.invalidateQueries({ queryKey: ['admin-products'] })
   174→        queryClient.invalidateQueries({ queryKey: ['stats'] })
   175→      } else {
   176→        toast.error(data.error || 'Erreur')
   177→      }
   178→    } catch {
   179→      toast.error('Erreur serveur')
   180→    }
   181→  }
   182→
   183→  const handleStockAdjust = async () => {
   184→    if (!stockAdjustProduct) return
   185→    const qty = parseInt(stockAdjustQty)
   186→    if (isNaN(qty) || qty < 0) { toast.error('Quantité invalide'); return }
   187→    setStockAdjustLoading(true)
   188→    try {
   189→      const res = await fetch(`/api/products/${stockAdjustProduct.id}`, {
   190→        method: 'PUT',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({ stockQty: qty, inStock: qty > 0 }),
   193→      })
   194→      if (res.ok) {
   195→        toast.success(`Stock mis à jour: ${qty} unités`)
   196→        setStockAdjustProduct(null)
   197→        queryClient.invalidateQueries({ queryKey: ['admin-products'] })
   198→      } else {
   199→        toast.error('Erreur lors de la mise à jour')
   200→      }
   201→    } catch {
   202→      toast.error('Erreur serveur')
   203→    } finally {
   204→      setStockAdjustLoading(false)
   205→    }
   206→  }
   207→
   208→  const toggleFeatured = async (product: Product) => {
   209→    try {
   210→      const res = await fetch(`/api/products/${product.id}`, {
   211→        method: 'PUT',
   212→        headers: { 'Content-Type': 'application/json' },
   213→        body: JSON.stringify({ isFeatured: !product.isFeatured }),
   214→      })
   215→      const data = await res.json()
   216→      if (data.success) {
   217→        toast.success(product.isFeatured ? 'Produit retiré des vedettes' : 'Produit mis en vedette')
   218→        queryClient.invalidateQueries({ queryKey: ['admin-products'] })
   219→      } else {
   220→        toast.error(data.error || 'Erreur')
   221→      }
   222→    } catch {
   223→      toast.error('Erreur serveur')
   224→    }
   225→  }
   226→
   227→  const toggleSelectProduct = (id: string) => {
   228→    setSelectedProducts(prev => {
   229→      const next = new Set(prev)
   230→      if (next.has(id)) next.delete(id)
   231→      else next.add(id)
   232→      return next
   233→    })
   234→  }
   235→
   236→  const toggleSelectAll = () => {
   237→    if (!products) return
   238→    if (selectedProducts.size === products.length) {
   239→      setSelectedProducts(new Set())
   240→    } else {
   241→      setSelectedProducts(new Set(products.map(p => p.id)))
   242→    }
   243→  }
   244→
   245→  const deleteSelectedProducts = async () => {
   246→    if (selectedProducts.size === 0) return
   247→    try {
   248→      await Promise.all(
   249→        [...selectedProducts].map(id => fetch(`/api/products/${id}`, { method: 'DELETE' }))
   250→      )
   251→      toast.success(`${selectedProducts.size} produit(s) supprimé(s)`)
   252→      setSelectedProducts(new Set())
   253→      queryClient.invalidateQueries({ queryKey: ['admin-products'] })
   254→      queryClient.invalidateQueries({ queryKey: ['stats'] })
   255→    } catch {
   256→      toast.error('Erreur lors de la suppression')
   257→    }
   258→  }
   259→
   260→  const exportSelectedCSV = () => {
   261→    if (!products || selectedProducts.size === 0) return
   262→    const selected = products.filter(p => selectedProducts.has(p.id))
   263→    const headers = ['Nom', 'Catégorie', 'Prix', 'Stock', 'Statut', 'Vedette']
   264→    const rows = selected.map(p => [
   265→      p.name,
   266→      p.category?.name || '',
   267→      String(p.price),
   268→      String(p.stockQty),
   269→      p.isActive ? 'Actif' : 'Inactif',
   270→      p.isFeatured ? 'Oui' : 'Non',
   271→    ])
   272→    const csv = [headers, ...rows].map(r => r.map(c => `"${c}"`).join(',')).join('\n')
   273→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   274→    const url = URL.createObjectURL(blob)
   275→    const a = document.createElement('a')
   276→    a.href = url
   277→    a.download = `produits-congoclean-${format(new Date(), 'yyyy-MM-dd')}.csv`
   278→    a.click()
   279→    URL.revokeObjectURL(url)
   280→    toast.success('Fichier CSV téléchargé')
   281→  }
   282→

   284→  const saveCategory = async () => {
   285→    if (!newCategoryName.trim()) { toast.error('Le nom est requis'); return }
   286→    setSaving(true)
   287→    try {
   288→      if (editingCategory) {
   289→        const slug = newCategorySlug || newCategoryName.toLowerCase().replace(/[^a-z0-9\s-]/g, '').replace(/\s+/g, '-').trim()
   290→        const res = await fetch(`/api/categories/${editingCategory.id}`, {
   291→          method: 'PUT',
   292→          headers: { 'Content-Type': 'application/json' },
   293→          body: JSON.stringify({ name: newCategoryName, slug }),
   294→        })
   295→        const data = await res.json()
   296→        if (data.success) { toast.success('Catégorie modifiée'); setCategoryDialogOpen(false); queryClient.invalidateQueries({ queryKey: ['categories'] }) }
   297→        else toast.error(data.error || 'Erreur')
   298→      } else {
   299→        const slug = newCategorySlug || newCategoryName.toLowerCase().replace(/[^a-z0-9\s-]/g, '').replace(/\s+/g, '-').trim()
   300→        const res = await fetch('/api/categories', {
   301→          method: 'POST',
   302→          headers: { 'Content-Type': 'application/json' },
   303→          body: JSON.stringify({ name: newCategoryName, slug }),
   304→        })
   305→        const data = await res.json()
   306→        if (data.success) { toast.success('Catégorie créée'); setCategoryDialogOpen(false); queryClient.invalidateQueries({ queryKey: ['categories'] }) }
   307→        else toast.error(data.error || 'Erreur')
   308→      }
   309→    } catch { toast.error('Erreur serveur') }
   310→    finally { setSaving(false) }
   311→  }
   312→
   313→  const deleteCategory = async () => {
   314→    if (!deletingCategory) return
   315→    try {
   316→      const res = await fetch(`/api/categories/${deletingCategory.id}`, { method: 'DELETE' })
   317→      const data = await res.json()
   318→      if (data.success) { toast.success('Catégorie supprimée'); queryClient.invalidateQueries({ queryKey: ['categories'] }) }
   319→      else toast.error(data.error || 'Erreur')
   320→    } catch { toast.error('Erreur serveur') }
   321→    finally { setDeleteCategoryDialogOpen(false); setDeletingCategory(null) }
   322→  }
   323→
   324→  return (
   325→    <div className="space-y-6">
   326→      {/* Top bar */}
   327→      <div className="flex flex-col sm:flex-row items-start sm:items-center gap-3">
   328→        <h3 className="text-xl font-semibold text-gray-900">Produits</h3>
   329→        <div className="flex-1" />
   330→        <div className="flex items-center gap-2 w-full sm:w-auto">
   331→          <div className="relative flex-1 sm:w-64">
   332→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   333→            <Input placeholder="Rechercher..." value={search} onChange={e => setSearch(e.target.value)} className="pl-9" />

   335→          <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   336→            <SelectTrigger className="w-[160px]"><SelectValue placeholder="Catégorie" /></SelectTrigger>
   337→            <SelectContent>
   338→              <SelectItem value="all">Toutes</SelectItem>
   339→              {categories?.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   340→            </SelectContent>
   341→          </Select>
   342→          <Button variant="outline" size="sm" onClick={() => setShowCategoryManager(!showCategoryManager)}>
   343→            <Filter className="h-4 w-4 mr-1" />Catégories
   344→          </Button>
   345→          <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => openProductDialog()}>
   346→            <Plus className="h-4 w-4 mr-1" />Ajouter
   347→          </Button>
   348→        </div>
   349→      </div>
   350→
   351→      {/* Category Manager */}
   352→      {showCategoryManager && (
   353→        <Card>
   354→          <CardHeader className="pb-3">
   355→            <div className="flex items-center justify-between">
   356→              <CardTitle className="text-base">Gestion des Catégories</CardTitle>
   357→              <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => { setEditingCategory(null); setNewCategoryName(''); setNewCategorySlug(''); setCategoryDialogOpen(true) }}>
   358→                <Plus className="h-4 w-4 mr-1" />Nouvelle Catégorie
   359→              </Button>
   360→            </div>
   361→          </CardHeader>
   362→          <CardContent>

   364→              <div className="space-y-2">

   367→                    <div>
   368→                      <p className="font-medium text-sm">{c.name}</p>
   369→                      <p className="text-xs text-muted-foreground">{c.slug} · {c._count.products} produit(s)</p>
   370→                    </div>
   371→                    <div className="flex items-center gap-1">
   372→                      <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => { setEditingCategory(c); setNewCategoryName(c.name); setNewCategorySlug(c.slug); setCategoryDialogOpen(true) }}>
   373→                        <Edit className="h-4 w-4" />
   374→                      </Button>
   375→                      <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:text-red-700" onClick={() => { setDeletingCategory(c); setDeleteCategoryDialogOpen(true) }}>
   376→                        <Trash2 className="h-4 w-4" />
   377→                      </Button>
   378→                    </div>
   379→                  </div>
   380→                ))}
   381→                {(!categories || categories.length === 0) && <p className="text-sm text-muted-foreground text-center py-4">Aucune catégorie</p>}
   382→              </div>
   383→            )}
   384→          </CardContent>
   385→        </Card>
   386→      )}
   387→
   388→      {/* Products table */}
   389→      <Card>
   390→        <CardContent className="p-0">
   391→          {productsLoading ? (
   392→            <div className="p-6 space-y-4"><Skeleton className="h-10 w-full" />{[1, 2, 3].map(i => <Skeleton key={i} className="h-16 w-full" />)}</div>
   393→          ) : products && products.length > 0 ? (
   394→            <div className="overflow-x-auto">
   395→              <Table>
   396→                <TableHeader>
   397→                  <TableRow className="bg-gray-50/80">
   398→                    <TableHead className="w-10">
   399→                      <input
   400→                        type="checkbox"
   401→                        checked={products.length > 0 && selectedProducts.size === products.length}
   402→                        onChange={toggleSelectAll}
   403→                        className="rounded border-gray-300 text-emerald-600 focus:ring-emerald-500 cursor-pointer"
   404→                      />
   405→                    </TableHead>
   406→                    <TableHead className="w-16">Image</TableHead>
   407→                    <TableHead>Nom</TableHead>
   408→                    <TableHead>Catégorie</TableHead>
   409→                    <TableHead className="text-right">Prix</TableHead>
   410→                    <TableHead className="text-right">Stock</TableHead>
   411→                    <TableHead>Vedette</TableHead>
   412→                    <TableHead>Statut</TableHead>
   413→                    <TableHead className="w-24">Actions</TableHead>
   414→                  </TableRow>
   415→                </TableHeader>
   416→                <TableBody>
   417→                  {products.map((p, i) => (
   418→                    <TableRow key={p.id} className={`${!p.isActive ? 'opacity-50' : ''} ${i % 2 === 0 ? 'bg-white' : 'bg-gray-50/30'} ${selectedProducts.has(p.id) ? 'bg-emerald-50/50' : ''}`}>
   419→                      <TableCell>
   420→                        <input
   421→                          type="checkbox"
   422→                          checked={selectedProducts.has(p.id)}
   423→                          onChange={() => toggleSelectProduct(p.id)}
   424→                          className="rounded border-gray-300 text-emerald-600 focus:ring-emerald-500 cursor-pointer"
   425→                        />
   426→                      </TableCell>
   427→                      <TableCell>
   428→                        <div className="w-10 h-10 rounded-lg overflow-hidden">
   429→                          {p.image ? (
   430→                            <img src={p.image} alt={p.name} className="w-full h-full object-cover" />
   431→                          ) : (
   432→                            <div className="w-full h-full rounded-lg bg-gradient-to-br from-emerald-100 to-teal-100 flex items-center justify-center">
   433→                              <Package className="h-4 w-4 text-emerald-300" />
   434→                            </div>
   435→                          )}
   436→                        </div>
   437→                      </TableCell>
   438→                      <TableCell>
   439→                        <div>
   440→                          <p className="font-medium text-sm">{p.name}</p>
   441→                          {p.volume && <p className="text-xs text-muted-foreground">{p.volume}</p>}
   442→                        </div>
   443→                      </TableCell>
   444→                      <TableCell className="text-sm">{p.category?.name}</TableCell>
   445→                      <TableCell className="text-right text-sm font-medium">{formatPrice(p.price)}</TableCell>
   446→                      <TableCell className="text-right">
   447→                        <button
   448→                          className={`text-sm font-medium hover:underline cursor-pointer ${p.stockQty < p.minStockAlert ? 'text-red-600 font-bold' : 'text-gray-900'}`}
   449→                          onClick={() => { setStockAdjustProduct(p); setStockAdjustQty(String(p.stockQty)) }}
   450→                        >
   451→                          {p.stockQty}
   452→                        </button>
   453→                      </TableCell>
   454→                      <TableCell>
   455→                        <button
   456→                          onClick={() => toggleFeatured(p)}
   457→                          className={`relative w-10 h-5 rounded-full transition-colors duration-200 ${p.isFeatured ? 'bg-emerald-500' : 'bg-gray-200'}`}
   458→                        >
   459→                          <span className={`absolute top-0.5 left-0.5 w-4 h-4 bg-white rounded-full shadow transition-transform duration-200 ${p.isFeatured ? 'translate-x-5' : 'translate-x-0'}`} />
   460→                        </button>
   461→                      </TableCell>
   462→                      <TableCell>
   463→                        <div className="flex items-center gap-2">
   464→                          <Badge variant="outline" className={p.isActive ? 'bg-emerald-50 text-emerald-700 border-emerald-200' : 'bg-gray-50 text-gray-500 border-gray-200'}>
   465→                            {p.isActive ? 'Actif' : 'Inactif'}
   466→                          </Badge>
   467→                          {p.isFeatured && <Badge variant="outline" className="bg-amber-50 text-amber-700 border-amber-200 text-xs">★</Badge>}
   468→                        </div>
   469→                      </TableCell>
   470→                      <TableCell>
   471→                        <div className="flex items-center gap-1">
   472→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openProductDialog(p)}><Edit className="h-4 w-4" /></Button>

   478→                          </Button>
   479→                        </div>
   480→                      </TableCell>
   481→                    </TableRow>
   482→                  ))}
   483→                </TableBody>
   484→              </Table>
   485→            </div>
   486→          ) : (
   487→            <div className="p-12 text-center">
   488→              <Package className="h-12 w-12 mx-auto text-gray-300 mb-3" />

   492→        </CardContent>

   494→
   495→      {selectedProducts.size > 0 && (
   496→        <div className="fixed bottom-6 left-1/2 -translate-x-1/2 z-50 flex items-center gap-3 bg-gray-900 text-white px-5 py-3 rounded-xl shadow-2xl">
   497→          <span className="text-sm font-medium">{selectedProducts.size} produit(s) sélectionné(s)</span>

   499→            <Trash2 className="h-4 w-4 mr-1" />
   500→            Supprimer la sélection
   501→          </Button>
   502→          <Button size="sm" variant="ghost" className="text-emerald-300 hover:text-emerald-200 hover:bg-emerald-900/50" onClick={exportSelectedCSV}>
   503→            <FileDown className="h-4 w-4 mr-1" />
   504→            Exporter CSV
   505→          </Button>
   506→        </div>
   507→      )}
   508→
   509→      {/* Product Dialog */}
   510→      <Dialog open={productDialogOpen} onOpenChange={setProductDialogOpen}>
   511→        <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   512→          <DialogHeader>
   513→            <DialogTitle>{editingProduct ? 'Modifier le Produit' : 'Ajouter un Produit'}</DialogTitle>
   514→            <DialogDescription>Remplissez les informations du produit</DialogDescription>
   515→          </DialogHeader>
   516→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 py-4">
   517→            <div className="sm:col-span-2 space-y-2">
   518→              <Label>Image du produit</Label>
   519→              <div className="relative group">
   520→                {editingProduct?.image && (
   521→                  <div className="mb-3 relative w-full h-40 rounded-lg overflow-hidden bg-gray-100">
   522→                    <img src={editingProduct.image} alt="Preview" className="w-full h-full object-cover" />

   524→                      type="button"
   525→                      onClick={() => {
   526→                        setEditingProduct(prev => prev ? { ...prev, image: null } : prev)

   528→                      className="absolute top-2 right-2 bg-red-500 text-white p-1 rounded-full opacity-0 group-hover:opacity-100 transition-opacity"
   529→                    >
   530→                      <X className="w-3 h-3" />
   531→                    </button>
   532→                  </div>
   533→                )}
   534→                <Input
   535→                  type="file"
   536→                  accept="image/jpeg,image/png,image/webp"
   537→                  onChange={async (e) => {

   539→                    if (!file) return
   540→                    setUploadingImage(true)
   541→                    try {
   542→                      const formData = new FormData()
   543→                      formData.append('image', file)
   544→                      const res = await fetch('/api/upload', { method: 'POST', body: formData })
   545→                      if (!res.ok) throw new Error('Upload failed')
   546→                      const data = await res.json()
   547→                      setEditingProduct(prev => prev ? { ...prev, image: data.data.url } : prev)
   548→                    } catch {
   549→                      toast.error('Erreur lors du téléchargement')
   550→                    } finally {
   551→                      setUploadingImage(false)
   552→                    }
   553→                  }}
   554→                  disabled={uploadingImage}
   555→                />
   556→                {uploadingImage && <p className="text-xs text-gray-500 mt-1">Téléchargement en cours...</p>}
   557→              </div>
   558→            </div>
   559→            <div className="sm:col-span-2 space-y-2">
   560→              <Label>Nom *</Label>
   561→              <Input value={pForm.name} onChange={e => setPForm({ ...pForm, name: e.target.value })} placeholder="Nom du produit" />
   562→            </div>
   563→            <div className="sm:col-span-2 space-y-2">
   564→              <Label>Catégorie *</Label>
   565→              <Select value={pForm.categoryId} onValueChange={v => setPForm({ ...pForm, categoryId: v })}>
   566→                <SelectTrigger><SelectValue placeholder="Sélectionner" /></SelectTrigger>
   567→                <SelectContent>{categories?.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}</SelectContent>
   568→              </Select>
   569→            </div>
   570→            <div className="sm:col-span-2 space-y-2">
   571→              <Label>Description</Label>
   572→              <Input value={pForm.description} onChange={e => setPForm({ ...pForm, description: e.target.value })} placeholder="Description courte" />
   573→            </div>
   574→            <div className="sm:col-span-2 space-y-2">
   575→              <Label>Description Longue</Label>
   576→              <Textarea value={pForm.longDescription} onChange={e => setPForm({ ...pForm, longDescription: e.target.value })} placeholder="Description détaillée" rows={3} />

   578→            <div className="space-y-2">
   579→              <Label>Prix (FCFA) *</Label>
   580→              <Input type="number" value={pForm.price} onChange={e => setPForm({ ...pForm, price: e.target.value })} placeholder="0" min="0" />

   582→            <div className="space-y-2">
   583→              <Label>Prix Comparé (FCFA)</Label>
   584→              <Input type="number" value={pForm.comparePrice} onChange={e => setPForm({ ...pForm, comparePrice: e.target.value })} placeholder="Optionnel" min="0" />
   585→            </div>
   586→            <div className="space-y-2">
   587→              <Label>Volume</Label>
   588→              <Input value={pForm.volume} onChange={e => setPForm({ ...pForm, volume: e.target.value })} placeholder="ex: 1L, 5L" />
   589→            </div>
   590→            <div className="space-y-2">
   591→              <Label>Stock</Label>
   592→              <Input type="number" value={pForm.stockQty} onChange={e => setPForm({ ...pForm, stockQty: e.target.value })} placeholder="0" min="0" />
   593→            </div>
   594→            <div className="space-y-2">
   595→              <Label>Alerte Stock Min</Label>
   596→              <Input type="number" value={pForm.minStockAlert} onChange={e => setPForm({ ...pForm, minStockAlert: e.target.value })} placeholder="10" min="0" />
   597→            </div>
   598→            <div className="flex items-center gap-8 pt-6">
   599→              <div className="flex items-center gap-2">
   600→                <Switch checked={pForm.isFeatured} onCheckedChange={v => setPForm({ ...pForm, isFeatured: v })} />
   601→                <Label>En vedette</Label>
   602→              </div>
   603→              <div className="flex items-center gap-2">
   604→                <Switch checked={pForm.isActive} onCheckedChange={v => setPForm({ ...pForm, isActive: v })} />
   605→                <Label>Actif</Label>
   606→              </div>
   607→            </div>
   608→          </div>
   609→          <DialogFooter>
   610→            <Button variant="outline" onClick={() => setProductDialogOpen(false)}>Annuler</Button>
   611→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={saveProduct} disabled={saving}>
   612→              {saving ? <RefreshCw className="h-4 w-4 animate-spin mr-2" /> : null}
   613→              {editingProduct ? 'Enregistrer' : 'Créer'}
   614→            </Button>
   615→          </DialogFooter>
   616→        </DialogContent>
   617→      </Dialog>
   618→
   619→      {/* Delete Product Dialog */}
   620→      <AlertDialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   621→        <AlertDialogContent>
   622→          <AlertDialogHeader>
   623→            <AlertDialogTitle>Supprimer le produit ?</AlertDialogTitle>
   624→            <AlertDialogDescription>
   625→              Êtes-vous sûr de vouloir supprimer &quot;{deletingProduct?.name}&quot; ? Cette action est irréversible.
   626→            </AlertDialogDescription>
   627→          </AlertDialogHeader>
   628→          <AlertDialogFooter>
   629→            <AlertDialogCancel>Annuler</AlertDialogCancel>
   630→            <AlertDialogAction className="bg-red-600 hover:bg-red-700" onClick={deleteProduct}>Supprimer</AlertDialogAction>
   631→          </AlertDialogFooter>
   632→        </AlertDialogContent>
   633→      </AlertDialog>
   634→
   635→      {/* Category Dialog */}
   636→      <Dialog open={categoryDialogOpen} onOpenChange={setCategoryDialogOpen}>
   637→        <DialogContent>
   638→          <DialogHeader>
   639→            <DialogTitle>{editingCategory ? 'Modifier la Catégorie' : 'Nouvelle Catégorie'}</DialogTitle>
   640→          </DialogHeader>
   641→          <div className="space-y-4 py-4">
   642→            <div className="space-y-2">
   643→              <Label>Nom *</Label>
   644→              <Input value={newCategoryName} onChange={e => { setNewCategoryName(e.target.value); if (!editingCategory) setNewCategorySlug(e.target.value.toLowerCase().replace(/[^a-z0-9\s-]/g, '').replace(/\s+/g, '-').trim()) }} />
   645→            </div>
   646→            <div className="space-y-2">
   647→              <Label>Slug</Label>
   648→              <Input value={newCategorySlug} onChange={e => setNewCategorySlug(e.target.value)} />
   649→            </div>
   650→          </div>
   651→          <DialogFooter>
   652→            <Button variant="outline" onClick={() => setCategoryDialogOpen(false)}>Annuler</Button>
   653→            <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={saveCategory} disabled={saving}>
   654→              {saving ? <RefreshCw className="h-4 w-4 animate-spin mr-2" /> : null}{editingCategory ? 'Enregistrer' : 'Créer'}
   655→            </Button>
   656→          </DialogFooter>
   657→        </DialogContent>
   658→      </Dialog>
   659→
   660→      {/* Delete Category Dialog */}
   661→      <AlertDialog open={deleteCategoryDialogOpen} onOpenChange={setDeleteCategoryDialogOpen}>
   662→        <AlertDialogContent>
   663→          <AlertDialogHeader>
   664→            <AlertDialogTitle>Supprimer la catégorie ?</AlertDialogTitle>
   665→            <AlertDialogDescription>
   666→              Êtes-vous sûr de vouloir supprimer &quot;{deletingCategory?.name}&quot; ? Les catégories avec des produits ne peuvent pas être supprimées.
   667→            </AlertDialogDescription>
   668→          </AlertDialogHeader>
   669→          <AlertDialogFooter>
   670→            <AlertDialogCancel>Annuler</AlertDialogCancel>
   671→            <AlertDialogAction className="bg-red-600 hover:bg-red-700" onClick={deleteCategory}>Supprimer</AlertDialogAction>
   672→          </AlertDialogFooter>
   673→        </AlertDialogContent>
   674→      </AlertDialog>
   675→
   676→      {/* Stock Adjustment Dialog */}

   678→        <DialogContent className="max-w-sm">
   679→          <DialogHeader>
   680→            <DialogTitle>Ajuster le stock</DialogTitle>
   681→            <DialogDescription>{stockAdjustProduct?.name}</DialogDescription>
   682→          </DialogHeader>
   683→          <div className="space-y-4">
   684→            <div>
   685→              <Label>Quantité en stock</Label>
   686→              <Input
   687→                type="number"
   688→                value={stockAdjustQty}
   689→                onChange={(e) => setStockAdjustQty(e.target.value)}
   690→                min="0"
   691→              />
   692→            </div>
   693→            <div className="flex gap-2">
   694→              {["+10", "+50", "+100", "-10", "Reset"].map(btn => (
   695→                <Button key={btn} variant="outline" size="sm" onClick={() => {
   696→                  if (btn === 'Reset') setStockAdjustQty(String(stockAdjustProduct?.stockQty || 0))
   697→                  else setStockAdjustQty(String(Math.max(0, parseInt(stockAdjustQty || '0') + parseInt(btn))))
   698→                }}>{btn}</Button>
   699→              ))}
   700→            </div>
   701→          </div>
   702→          <DialogFooter>
   703→            <Button variant="outline" onClick={() => setStockAdjustProduct(null)}>Annuler</Button>
   704→            <Button onClick={handleStockAdjust} disabled={stockAdjustLoading}>
   705→              {stockAdjustLoading ? <RefreshCw className="h-4 w-4 animate-spin mr-2" /> : null}
   706→              Enregistrer
   707→            </Button>
