   332→   332→function PartnerDetail({ partnerId, onBack }: { partnerId: string; onBack: () => void }) {
   333→   333→  const [partner, setPartner] = useState<Affiliate | null>(null)
   334→   334→  const [commissions, setCommissions] = useState<Commission[]>([])
   335→   335→  const [payouts, setPayouts] = useState<AffiliatePayout[]>([])

   337→   337→  const [commissionPage, setCommissionPage] = useState(1)
   338→   338→  const [commissionTotal, setCommissionTotal] = useState(0)
   339→   339→  const [commissionTotalPages, setCommissionTotalPages] = useState(1)
   340→   340→  const COMMISSIONS_PER_PAGE = 20
   341→   341→
   342→   342→  // Edit mode
   343→   343→  const [editing, setEditing] = useState(false)
   344→   344→  const [editForm, setEditForm] = useState({ name: '', email: '', phone: '', company: '', commissionRate: '5', bankInfo: '', isActive: true })
   345→   345→  const [editLoading, setEditLoading] = useState(false)
   346→   346→
   347→   347→  // Payout dialog
   348→   348→  const [payoutOpen, setPayoutOpen] = useState(false)
   349→   349→  const [payoutForm, setPayoutForm] = useState({ amount: '', method: 'mobile_money', reference: '' })
   350→   350→  const [payoutLoading, setPayoutLoading] = useState(false)
   351→   351→
   352→   352→  // Commission breakdown
   353→   353→  const [breakdown, setBreakdown] = useState({ pending: 0, validated: 0, paid: 0, cancelled: 0 })
   354→   354→
   355→   355→  const fetchData = useCallback(async () => {
   356→   356→    try {
   357→   357→      setLoading(true)
   358→   358→      const [partnerRes, commRes, payoutRes] = await Promise.all([
   359→   359→        fetch(`/api/admin/partners?id=${partnerId}`),
   360→   360→        fetch(`/api/commissions?affiliateId=${partnerId}&page=${commissionPage}&limit=${COMMISSIONS_PER_PAGE}`),
   361→   361→        fetch(`/api/admin/partners?payouts=1&affiliateId=${partnerId}`),
   362→   362→      ])
   363→   363→
   364→   364→      if (partnerRes.ok) {
   365→   365→        const pData = await partnerRes.json()
   366→   366→        setPartner(pData.partner || null)
   367→   367→        if (pData.partner) {
   368→   368→          setEditForm({
   369→   369→            name: pData.partner.name,
   370→   370→            email: pData.partner.email,
   371→   371→            phone: pData.partner.phone || '',
   372→   372→            company: pData.partner.company || '',
   373→   373→            commissionRate: String(pData.partner.commissionRate),
   374→   374→            bankInfo: pData.partner.bankInfo || '',
   375→   375→            isActive: pData.partner.isActive,
   376→   376→          })
   377→   377→        }
   378→   378→      }
   379→   379→
   380→   380→      if (commRes.ok) {
   381→   381→        const cData = await commRes.json()
   382→   382→        setCommissions(cData.data?.commissions || [])
   383→   383→        setCommissionTotal(cData.data?.total || 0)
   384→   384→        setCommissionTotalPages(cData.data?.totalPages || 1)
   385→   385→        // Compute breakdown from summary
   386→   386→        const summary = cData.data?.summary || []
   387→   387→        const bd = { pending: 0, validated: 0, paid: 0, cancelled: 0 }
   388→   388→        for (const s of summary) {
   389→   389→          if (s.status in bd) bd[s.status as keyof typeof bd] = s.total
   390→   390→        }
   391→   391→        setBreakdown(bd)
   392→   392→      }
   393→   393→
   394→   394→      if (payoutRes.ok) {
   395→   395→        const payData = await payoutRes.json()
   396→   396→        setPayouts(payData.payouts || [])
   397→   397→      }
   398→   398→    } catch {
   399→   399→      toast.error('Erreur de chargement')
   400→   400→    } finally {
   401→   401→      setLoading(false)
   402→   402→    }
   403→   403→  }, [partnerId, commissionPage])
   404→   404→
   405→   405→  useEffect(() => { fetchData() }, [fetchData])
   406→   406→
   407→   407→  const handleSaveEdit = async () => {
   408→   408→    try {
   409→   409→      setEditLoading(true)
   410→   410→      const res = await fetch('/api/admin/partners', {
   411→   411→        method: 'PUT',
   412→   412→        headers: { 'Content-Type': 'application/json' },
   413→   413→        body: JSON.stringify({
   414→   414→          id: partnerId,
   415→   415→          name: editForm.name,
   416→   416→          email: editForm.email,
   417→   417→          phone: editForm.phone,
   418→   418→          company: editForm.company,
   419→   419→          commissionRate: parseFloat(editForm.commissionRate) || 5,
   420→   420→          bankInfo: editForm.bankInfo,
   421→   421→          isActive: editForm.isActive,
   422→   422→        }),
   423→   423→      })
