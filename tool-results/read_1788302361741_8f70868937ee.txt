   171→   171→              {s.label}
   172→   172→            </span>
   173→   173→          </div>
   174→   174→          {i < steps.length - 1 && (
   175→   175→            <div className={`flex-1 h-0.5 mx-3 rounded-full transition-colors duration-300 ${step > s.num ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   176→   176→          )}
   177→   177→        </div>
