   100→   100→export function CustomerDashboard({ open, onOpenChange, onLogout }: CustomerDashboardProps) {
   101→   101→  const { customer, logout, isAuthenticated } = useCustomerAuthStore()
   102→   102→
   103→   103→  // Orders state
   104→   104→  const [orders, setOrders] = useState<Order[]>([])
   105→   105→  const [ordersLoading, setOrdersLoading] = useState(true)
   106→   106→  const [expandedOrder, setExpandedOrder] = useState<string | null>(null)
   107→   107→  const [page, setPage] = useState(1)
   108→   108→  const [totalPages, setTotalPages] = useState(1)
   109→   109→  const [totalOrders, setTotalOrders] = useState(0)
   110→   110→  const ordersPerPage = 5
   111→   111→
   112→   112→  // Chat state
   113→   113→  const [chatMessages, setChatMessages] = useState<ChatMessage[]>([])
   114→   114→  const [chatInput, setChatInput] = useState('')
   115→   115→  const [chatLoading, setChatLoading] = useState(false)
   116→   116→  const [chatSending, setChatSending] = useState(false)
   117→   117→  const [showChat, setShowChat] = useState(false)
   118→   118→  const chatEndRef = useRef<HTMLDivElement>(null)
   119→   119→
   120→   120→  // Active orders count
   121→   121→  const activeOrdersCount = orders.filter(
   122→   122→    (o) => o.status === 'pending' || o.status === 'confirmed' || o.status === 'processing' || o.status === 'shipped'
   123→   123→  ).length
   124→   124→
   125→   125→  // Fetch orders
   126→   126→  const fetchOrders = useCallback(async (p: number) => {
   127→   127→    setOrdersLoading(true)
   128→   128→    try {
   129→   129→      const res = await fetch('/api/customer-auth', {
   130→   130→        method: 'POST',
   131→   131→        headers: { 'Content-Type': 'application/json' },
   132→   132→        body: JSON.stringify({ action: 'orders', page: p, limit: ordersPerPage }),
   133→   133→      })
   134→   134→      if (res.ok) {
   135→   135→        const data = await res.json()
   136→   136→        const ordersData: Order[] = data.orders || data.data || []
   137→   137→        setOrders(ordersData)
   138→   138→        setTotalPages(data.totalPages || Math.ceil((data.total || ordersData.length) / ordersPerPage) || 1)
   139→   139→        setTotalOrders(data.total || ordersData.length)
   140→   140→      } else {
   141→   141→        setOrders([])
   142→   142→      }
   143→   143→    } catch {
   144→   144→      setOrders([])
   145→   145→    } finally {
   146→   146→      setOrdersLoading(false)
   147→   147→    }
   148→   148→  }, [])
   149→   149→
   150→   150→  useEffect(() => {
   151→   151→    if (isAuthenticated) {
   152→   152→      fetchOrders(page)
   153→   153→    }
   154→   154→  }, [isAuthenticated, page, fetchOrders])
   155→   155→
   156→   156→  // --- Chat logic ---
   157→   157→  const chatSessionId = customer ? `customer-${customer.id}` : ''
   158→   158→
   159→   159→  const fetchChatMessages = useCallback(async () => {
   160→   160→    if (!chatSessionId) return
   161→   161→    try {
   162→   162→      const res = await fetch(`/api/chat?sessionId=${encodeURIComponent(chatSessionId)}`)
   163→   163→      if (res.ok) {
   164→   164→        const data = await res.json()
   165→   165→        const msgs: ChatMessage[] = data?.data?.messages || []
   166→   166→        setChatMessages(msgs)
   167→   167→      }
   168→   168→    } catch {
   169→   169→      // silently ignore polling errors
   170→   170→    }
   171→   171→  }, [chatSessionId])
   172→   172→
   173→   173→  const loadChat = useCallback(async () => {
   174→   174→    if (!chatSessionId) return
   175→   175→    setChatLoading(true)
   176→   176→    try {
   177→   177→      const res = await fetch(`/api/chat?sessionId=${encodeURIComponent(chatSessionId)}`)
   178→   178→      if (res.ok) {
   179→   179→        const data = await res.json()
   180→   180→        const msgs: ChatMessage[] = data?.data?.messages || []
   181→   181→        setChatMessages(msgs)
   182→   182→      } else {
   183→   183→        setChatMessages([])
   184→   184→      }
   185→   185→    } catch {
   186→   186→      setChatMessages([])
   187→   187→    } finally {
   188→   188→      setChatLoading(false)
   189→   189→    }
   190→   190→  }, [chatSessionId])
   191→   191→
   192→   192→  const sendChatMessage = async () => {
   193→   193→    const trimmed = chatInput.trim()
   194→   194→    if (!trimmed || chatSending || !chatSessionId) return
   195→   195→    setChatSending(true)
   196→   196→    try {
   197→   197→      const res = await fetch('/api/chat', {
   198→   198→        method: 'POST',
   199→   199→        headers: { 'Content-Type': 'application/json' },
   200→   200→        body: JSON.stringify({ sessionId: chatSessionId, content: trimmed }),
   201→   201→      })
   202→   202→      if (res.ok) {
   203→   203→        setChatInput('')
   204→   204→        // Re-fetch to get the full list including the new message
   205→   205→        await fetchChatMessages()
   206→   206→      } else {
   207→   207→        toast.error('Erreur lors de l\'envoi du message')
   208→   208→      }
   209→   209→    } catch {
   210→   210→      toast.error('Erreur lors de l\'envoi du message')
   211→   211→    } finally {
   212→   212→      setChatSending(false)
   213→   213→    }
   214→   214→  }
   215→   215→
   216→   216→  // Load chat on open & poll for new messages
   217→   217→  useEffect(() => {
   218→   218→    if (showChat && chatSessionId) {
   219→   219→      loadChat()
   220→   220→      const interval = setInterval(() => {
   221→   221→        fetchChatMessages()
   222→   222→      }, 5000)
   223→   223→      return () => clearInterval(interval)
   224→   224→    }
   225→   225→  }, [showChat, chatSessionId, loadChat, fetchChatMessages])
   226→   226→
   227→   227→  // Auto-scroll chat to bottom
   228→   228→  useEffect(() => {
   229→   229→    if (showChat && chatMessages.length > 0) {
   230→   230→      chatEndRef.current?.scrollIntoView({ behavior: 'smooth' })
   231→   231→    }
   232→   232→  }, [showChat, chatMessages])
   233→   233→
   234→   234→  const formatChatTime = (dateStr: string) => {
   235→   235→    try {
   236→   236→      return new Date(dateStr).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit' })
   237→   237→    } catch {
   238→   238→      return ''
   239→   239→    }
   240→   240→  }
   241→   241→
   242→   242→  const lastAdminMessage = chatMessages.filter((m) => m.senderType === 'admin').slice(-1)[0]
   243→   243→
   244→   244→  // Logout handler
   245→   245→  const handleLogout = async () => {
   246→   246→    await logout()
   247→   247→    toast.success('Déconnexion réussie')
   248→   248→    onLogout?.()
   249→   249→  }
   250→   250→
   251→   251→  // Toggle order details
   252→   252→  const toggleExpand = (orderId: string) => {
   253→   253→    setExpandedOrder((prev) => (prev === orderId ? null : orderId))
   254→   254→  }
   255→   255→
   256→   256→  // Format date
   257→   257→  const formatDate = (dateStr: string) => {
   258→   258→    try {
   259→   259→      return new Date(dateStr).toLocaleDateString('fr-FR', {
   260→   260→        day: 'numeric',
   261→   261→        month: 'long',
   262→   262→        year: 'numeric',
   263→   263→      })
   264→   264→    } catch {
   265→   265→      return dateStr
   266→   266→    }
   267→   267→  }
   268→   268→
   269→   269→  if (!customer) return null
   270→   270→
   271→   271→  return (
   272→   272→    <Dialog open={open} onOpenChange={onOpenChange}>
   273→   273→      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto p-0">
   274→   274→        <DialogHeader className="px-6 pt-6 pb-0 sr-only">
   275→   275→          <DialogTitle>Mon Compte</DialogTitle>
   276→   276→        </DialogHeader>
   277→   277→    <div className="w-full max-w-4xl mx-auto space-y-6 px-6 py-6">
   278→   278→      {/* Header */}
   279→   279→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   280→   280→        <div className="flex items-center gap-3">
   281→   281→          <div className="w-12 h-12 rounded-full bg-[#1a1a1a] flex items-center justify-center">
   282→   282→            <User className="w-6 h-6 text-white" />
   283→   283→          </div>
   284→   284→          <div>
   285→   285→            <h2 className="text-xl font-bold text-[#1a1a1a] tracking-tight">Mon Compte</h2>
   286→   286→            <p className="text-sm text-[#888888]">{customer.name}</p>
   287→   287→          </div>
   288→   288→        </div>
   289→   289→        <Button
   290→   290→          variant="outline"
   291→   291→          onClick={handleLogout}
   292→   292→          className="w-fit border-gray-200 text-[#1a1a1a] hover:bg-red-50 hover:text-red-600 hover:border-red-200 rounded-lg transition-colors duration-200"
   293→   293→        >
   294→   294→          <LogOut className="w-4 h-4 mr-2" />
   295→   295→          Déconnexion
   296→   296→        </Button>
   297→   297→      </div>
   298→   298→
   299→   299→      {/* Stats Cards */}
   300→   300→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   301→   301→        {/* Total commandes */}
   302→   302→        <Card className="border-gray-200 shadow-sm rounded-xl">
   303→   303→          <CardContent className="p-5">
   304→   304→            <div className="flex items-center justify-between">
   305→   305→              <div>
   306→   306→                <p className="text-xs font-medium text-[#888888] uppercase tracking-wider">Total commandes</p>
   307→   307→                <p className="text-2xl font-bold text-[#1a1a1a] mt-1">{customer.ordersCount || totalOrders}</p>
   308→   308→              </div>
   309→   309→              <div className="w-10 h-10 rounded-lg bg-gray-100 flex items-center justify-center">
   310→   310→                <Package className="w-5 h-5 text-[#1a1a1a]" />
   311→   311→              </div>
   312→   312→            </div>
   313→   313→          </CardContent>
   314→   314→        </Card>
   315→   315→
   316→   316→        {/* Total dépensé */}
   317→   317→        <Card className="border-gray-200 shadow-sm rounded-xl">
   318→   318→          <CardContent className="p-5">
   319→   319→            <div className="flex items-center justify-between">
   320→   320→              <div>
   321→   321→                <p className="text-xs font-medium text-[#888888] uppercase tracking-wider">Total dépensé</p>
   322→   322→                <p className="text-2xl font-bold text-[#1a1a1a] mt-1">{formatPrice(customer.totalSpent || 0)}</p>
   323→   323→              </div>
   324→   324→              <div className="w-10 h-10 rounded-lg bg-gray-100 flex items-center justify-center">
   325→   325→                <DollarSign className="w-5 h-5 text-[#1a1a1a]" />
   326→   326→              </div>
   327→   327→            </div>
   328→   328→          </CardContent>
   329→   329→        </Card>
   330→   330→
   331→   331→        {/* Commandes en cours */}
   332→   332→        <Card className="border-gray-200 shadow-sm rounded-xl">
   333→   333→          <CardContent className="p-5">
   334→   334→            <div className="flex items-center justify-between">
   335→   335→              <div>
   336→   336→                <p className="text-xs font-medium text-[#888888] uppercase tracking-wider">En cours</p>
   337→   337→                <p className="text-2xl font-bold text-[#1a1a1a] mt-1">{activeOrdersCount}</p>
   338→   338→              </div>
   339→   339→              <div className="w-10 h-10 rounded-lg bg-gray-100 flex items-center justify-center">
   340→   340→                <Clock className="w-5 h-5 text-[#1a1a1a]" />
   341→   341→              </div>
   342→   342→            </div>
   343→   343→          </CardContent>
   344→   344→        </Card>
   345→   345→      </div>
   346→   346→
   347→   347→      {/* Customer Info Card */}
   348→   348→      <Card className="border-gray-200 shadow-sm rounded-xl">
   349→   349→        <CardHeader className="pb-2">
   350→   350→          <CardTitle className="text-base font-semibold text-[#1a1a1a] flex items-center gap-2">
   351→   351→            <User className="w-4 h-4 text-[#888888]" />
   352→   352→            Mes informations
   353→   353→          </CardTitle>
   354→   354→        </CardHeader>
   355→   355→        <CardContent className="pb-5">
   356→   356→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 text-sm">
   357→   357→            <div className="flex items-center gap-2">
   358→   358→              <span className="text-[#888888]">Email :</span>
   359→   359→              <span className="text-[#1a1a1a] font-medium">{customer.email}</span>
   360→   360→            </div>
   361→   361→            <div className="flex items-center gap-2">
   362→   362→              <span className="text-[#888888]">Tél :</span>
   363→   363→              <span className="text-[#1a1a1a] font-medium">{customer.phone}</span>
   364→   364→            </div>
   365→   365→            <div className="flex items-center gap-2">
   366→   366→              <MapPin className="w-3.5 h-3.5 text-[#888888] shrink-0" />
   367→   367→              <span className="text-[#1a1a1a] font-medium">
   368→   368→                {customer.address}, {customer.quartier}, {customer.city}
   369→   369→              </span>
   370→   370→            </div>
   371→   371→          </div>
   372→   372→        </CardContent>
   373→   373→      </Card>
   374→   374→
   375→   375→      {/* Chat with store button */}
   376→   376→      {!showChat && (
   377→   377→        <button
   378→   378→          onClick={() => setShowChat(true)}
   379→   379→          className="w-full flex items-center gap-4 p-5 border border-gray-200 rounded-xl shadow-sm hover:shadow-md transition-shadow duration-200 bg-white"
   380→   380→        >
   381→   381→          <div className="w-10 h-10 rounded-lg bg-[#1a1a1a] flex items-center justify-center shrink-0">
   382→   382→            <MessageCircle className="w-5 h-5 text-white" />
   383→   383→          </div>
   384→   384→          <div className="text-left flex-1">
   385→   385→            <p className="text-sm font-semibold text-[#1a1a1a]">Discuter avec la boutique</p>
   386→   386→            <p className="text-xs text-[#888888] mt-0.5">Posez vos questions, suivez vos commandes en direct</p>
   387→   387→          </div>
   388→   388→          <MessageCircle className="w-4 h-4 text-[#888888] shrink-0" />
   389→   389→        </button>
   390→   390→      )}
   391→   391→
   392→   392→      {/* Chat view */}
   393→   393→      {showChat && (
   394→   394→        <Card className="border-gray-200 shadow-sm rounded-xl overflow-hidden">
   395→   395→          {/* Chat header */}
   396→   396→          <div className="flex items-center justify-between px-4 py-3 bg-[#1a1a1a]">
   397→   397→            <div className="flex items-center gap-3">
   398→   398→              <button
   399→   399→                onClick={() => setShowChat(false)}
   400→   400→                className="w-7 h-7 flex items-center justify-center text-white/60 hover:text-white rounded-full hover:bg-white/10 transition-colors"
   401→   401→                aria-label="Retour au tableau de bord"
   402→   402→              >
   403→   403→                <ArrowLeft className="w-4 h-4" />
   404→   404→              </button>
   405→   405→              <div>
   406→   406→                <p className="text-white text-sm font-medium leading-tight">KALA'S</p>
   407→   407→                <p className="text-white/60 text-xs">Discuter avec la boutique</p>
   408→   408→              </div>
   409→   409→            </div>
   410→   410→            {/* Admin online indicator */}
   411→   411→            <div className="flex items-center gap-1.5">
   412→   412→              <span className="w-2 h-2 rounded-full bg-green-400" />
   413→   413→              <span className="text-white/60 text-xs">
   414→   414→                {lastAdminMessage
   415→   415→                  ? `Rép. à ${formatChatTime(lastAdminMessage.createdAt)}`
   416→   416→                  : 'En ligne'}
   417→   417→              </span>
   418→   418→            </div>
   419→   419→          </div>
   420→   420→
   421→   421→          {/* Messages area */}
   422→   422→          <div className="h-[320px] overflow-y-auto px-4 py-3 space-y-3">
   423→   423→            {chatLoading ? (
   424→   424→              <div className="flex items-center justify-center py-12">

   427→   427→              </div>
   428→   428→            ) : chatMessages.length === 0 ? (
   429→   429→              <div className="flex flex-col items-center justify-center py-10">
   430→   430→                <MessageCircle className="w-10 h-10 text-[#888888] mb-3 opacity-30" />
   431→   431→                <p className="text-sm text-[#888888] text-center">Aucun message pour le moment</p>
   432→   432→                <p className="text-xs text-[#888888] mt-1 text-center">Envoyez un message pour commencer la conversation</p>
   433→   433→              </div>
   434→   434→            ) : (
   435→   435→              chatMessages.map((msg) => (
   436→   436→                <div
   437→   437→                  key={msg.id}
   438→   438→                  className={msg.senderType === 'customer' ? 'flex justify-end' : 'flex justify-start'}
   439→   439→                >
   440→   440→                  <div
   441→   441→                    className={
   442→   442→                      msg.senderType === 'customer'
   443→   443→                        ? 'bg-[#1a1a1a] text-white px-3.5 py-2.5 rounded-2xl rounded-br-md max-w-[80%] text-sm leading-relaxed'
