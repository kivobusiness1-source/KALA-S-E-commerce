   332→function PartnerDetail({ partnerId, onBack }: { partnerId: string; onBack: () => void }) {
   333→  const [partner, setPartner] = useState<Affiliate | null>(null)
   334→  const [commissions, setCommissions] = useState<Commission[]>([])
   335→  const [payouts, setPayouts] = useState<AffiliatePayout[]>([])
   336→  const [loading, setLoading] = useState(true)
   337→  const [commissionPage, setCommissionPage] = useState(1)
   338→  const [commissionTotal, setCommissionTotal] = useState(0)
   339→  const [commissionTotalPages, setCommissionTotalPages] = useState(1)
   340→  const COMMISSIONS_PER_PAGE = 20
   341→
   342→  // Edit mode
   343→  const [editing, setEditing] = useState(false)
   344→  const [editForm, setEditForm] = useState({ name: '', email: '', phone: '', company: '', commissionRate: '5', bankInfo: '', isActive: true })
   345→  const [editLoading, setEditLoading] = useState(false)
   346→
   347→  // Payout dialog
   348→  const [payoutOpen, setPayoutOpen] = useState(false)
   349→  const [payoutForm, setPayoutForm] = useState({ amount: '', method: 'mobile_money', reference: '' })
   350→  const [payoutLoading, setPayoutLoading] = useState(false)
   351→
   352→  // Commission breakdown
   353→  const [breakdown, setBreakdown] = useState({ pending: 0, validated: 0, paid: 0, cancelled: 0 })
   354→
   355→  const fetchData = useCallback(async () => {
   356→    try {
   357→      setLoading(true)
   358→      const [partnerRes, commRes, payoutRes] = await Promise.all([
   359→        fetch(`/api/admin/partners?id=${partnerId}`),
   360→        fetch(`/api/commissions?affiliateId=${partnerId}&page=${commissionPage}&limit=${COMMISSIONS_PER_PAGE}`),
   361→        fetch(`/api/admin/partners?payouts=1&affiliateId=${partnerId}`),
   362→      ])
   363→
   364→      if (partnerRes.ok) {
   365→        const pData = await partnerRes.json()
   366→        setPartner(pData.partner || null)
   367→        if (pData.partner) {
   368→          setEditForm({
   369→            name: pData.partner.name,
   370→            email: pData.partner.email,
   371→            phone: pData.partner.phone || '',
   372→            company: pData.partner.company || '',
   373→            commissionRate: String(pData.partner.commissionRate),
   374→            bankInfo: pData.partner.bankInfo || '',
   375→            isActive: pData.partner.isActive,
   376→          })
   377→        }
   378→      }
   379→
   380→      if (commRes.ok) {
   381→        const cData = await commRes.json()
   382→        setCommissions(cData.data?.commissions || [])
   383→        setCommissionTotal(cData.data?.total || 0)
   384→        setCommissionTotalPages(cData.data?.totalPages || 1)
   385→        // Compute breakdown from summary
   386→        const summary = cData.data?.summary || []
   387→        const bd = { pending: 0, validated: 0, paid: 0, cancelled: 0 }
   388→        for (const s of summary) {
   389→          if (s.status in bd) bd[s.status as keyof typeof bd] = s.total
   390→        }
   391→        setBreakdown(bd)
   392→      }
   393→
   394→      if (payoutRes.ok) {
   395→        const payData = await payoutRes.json()
   396→        setPayouts(payData.payouts || [])
   397→      }
   398→    } catch {
   399→      toast.error('Erreur de chargement')
   400→    } finally {
   401→      setLoading(false)
   402→    }
   403→  }, [partnerId, commissionPage])
   404→
   405→  useEffect(() => { fetchData() }, [fetchData])
   406→
   407→  const handleSaveEdit = async () => {
   408→    try {
   409→      setEditLoading(true)
   410→      const res = await fetch('/api/admin/partners', {
   411→        method: 'PUT',
   412→        headers: { 'Content-Type': 'application/json' },
   413→        body: JSON.stringify({
   414→          id: partnerId,
   415→          name: editForm.name,
   416→          email: editForm.email,
   417→          phone: editForm.phone,
   418→          company: editForm.company,
   419→          commissionRate: parseFloat(editForm.commissionRate) || 5,
   420→          bankInfo: editForm.bankInfo,
   421→          isActive: editForm.isActive,
   422→        }),
   423→      })
   424→      const data = await res.json()
   425→      if (res.ok && data.success) {
   426→        toast.success('Partenaire mis à jour')
   427→        setEditing(false)
   428→        fetchData()
   429→      } else {
   430→        toast.error(data.error || 'Erreur')
   431→      }
   432→    } catch {
   433→      toast.error('Erreur serveur')
   434→    } finally {
   435→      setEditLoading(false)
   436→    }
   437→  }
   438→
   439→  const handleCommissionAction = async (commissionId: string, status: CommissionStatus) => {
   440→    try {
   441→      const res = await fetch('/api/commissions', {
   442→        method: 'PUT',
   443→        headers: { 'Content-Type': 'application/json' },
   444→        body: JSON.stringify({ id: commissionId, status }),
   445→      })
   446→      const data = await res.json()
   447→      if (res.ok && data.success) {
   448→        toast.success(`Commission ${COMMISSION_STATUS_LABELS[status]}`)
   449→        fetchData()
   450→      } else {
   451→        toast.error(data.error || 'Erreur')
   452→      }
   453→    } catch {
   454→      toast.error('Erreur serveur')
   455→    }
   456→  }
   457→
   458→  const handleCreatePayout = async () => {
   459→    const amount = parseFloat(payoutForm.amount)
   460→    if (!amount || amount <= 0) {
   461→      toast.error('Montant invalide')
   462→      return
   463→    }
   464→    try {
   465→      setPayoutLoading(true)
   466→      const res = await fetch('/api/admin/partners', {
   467→        method: 'PUT',
   468→        headers: { 'Content-Type': 'application/json' },
   469→        body: JSON.stringify({
   470→          id: partnerId,
   471→          action: 'createPayout',
   472→          payoutAmount: amount,
   473→          payoutMethod: payoutForm.method,
   474→          payoutReference: payoutForm.reference,
   475→        }),
   476→      })
   477→      const data = await res.json()
   478→      if (res.ok && data.success) {
   479→        toast.success('Paiement créé avec succès')
   480→        setPayoutOpen(false)
   481→        setPayoutForm({ amount: '', method: 'mobile_money', reference: '' })
   482→        fetchData()
   483→      } else {
   484→        toast.error(data.error || 'Erreur')
   485→      }
   486→    } catch {
   487→      toast.error('Erreur serveur')
   488→    } finally {
   489→      setPayoutLoading(false)
   490→    }
   491→  }
   492→
   493→  if (loading) {
   494→    return (
   495→      <div className="flex items-center justify-center h-64">
   496→        <Loader2 className="w-8 h-8 animate-spin text-gray-400" />
   497→      </div>
   498→    )
   499→  }
   500→
   501→  if (!partner) {
   502→    return (
   503→      <div className="text-center py-12 text-gray-400">
   504→        <p>Partenaire introuvable</p>
   505→        <Button variant="outline" onClick={onBack} className="mt-4">Retour</Button>
   506→      </div>
   507→    )
   508→  }
   509→
   510→  return (
   511→    <div className="space-y-6">
   512→      {/* Back button */}
   513→      <Button variant="ghost" onClick={onBack} className="gap-1.5 text-gray-600">
   514→        <ArrowLeft className="w-4 h-4" />
   515→        Retour à la liste
   516→      </Button>
   517→
   518→      {/* Partner Info Card */}
   519→      <Card>
   520→        <CardHeader>
   521→          <div className="flex items-center justify-between">
   522→            <CardTitle className="flex items-center gap-2 text-base">
   523→              <Handshake className="w-5 h-5 text-emerald-600" />
