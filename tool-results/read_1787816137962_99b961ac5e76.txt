   103→function getCategoryBorderColor(slug: string): string {
   104→  switch (slug) {
   105→    case 'savon-liquide':
   106→      return 'border-l-emerald-500'
   107→    case 'detergent':
   108→      return 'border-l-amber-500'
   109→    case 'eau-de-javel':
   110→      return 'border-l-cyan-500'
   111→    default:
   112→      return 'border-l-emerald-500'
   113→  }

   147→function OrderProgressIndicator({ step }: { step: 1 | 2 | 3 }) {
   148→  const steps = [
   149→    { num: 1, label: 'Informations' },
   150→    { num: 2, label: 'Livraison' },
   151→    { num: 3, label: 'Confirmation' },
   152→  ]
   153→  return (
   154→    <div className="flex items-center justify-between mb-6">
   155→      {steps.map((s, i) => (
   156→        <div key={s.num} className="flex items-center flex-1 last:flex-none">
   157→          <div className="flex flex-col items-center gap-1.5">
   158→            <div
   159→              className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-semibold transition-colors duration-300 ${
   160→                step >= s.num
   161→                  ? 'bg-emerald-600 text-white shadow-md shadow-emerald-600/25'
   162→                  : 'bg-gray-100 text-gray-400'
   163→              }`}
   164→            >
   165→              {step > s.num ? <Check className="w-4 h-4" /> : s.num}
   166→            </div>
   167→            <span className={`text-[11px] font-medium ${step >= s.num ? 'text-emerald-700' : 'text-gray-400'}`}>
   168→              {s.label}
   169→            </span>
   170→          </div>
   171→          {i < steps.length - 1 && (
   172→            <div className={`flex-1 h-0.5 mx-3 rounded-full transition-colors duration-300 ${step > s.num ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   173→          )}
   174→        </div>
   175→      ))}
   176→    </div>
   177→  )
   178→}
