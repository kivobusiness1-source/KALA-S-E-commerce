   195→   195→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   196→   196→  const [count, setCount] = useState(0)
   197→   197→  const ref = useRef(null)
   198→   198→  const isInView = useInView(ref, { once: true })
   199→   199→
   200→   200→  useEffect(() => {
   201→   201→    if (!isInView) return
   202→   202→    let start = 0
   203→   203→    const duration = 2000
   204→   204→    const step = target / (duration / 16)
   205→   205→    const timer = setInterval(() => {
   206→   206→      start += step
   207→   207→      if (start >= target) {
   208→   208→        setCount(target)
   209→   209→        clearInterval(timer)
   210→   210→      } else {
   211→   211→        setCount(Math.floor(start))
   212→   212→      }
   213→   213→    }, 16)
   214→   214→    return () => clearInterval(timer)
   215→   215→  }, [isInView, target])
   216→   216→
   217→   217→  return <span ref={ref}>{count}{suffix}</span>
   218→   218→}

   222→   222→function TypingEffect({ phrases }: { phrases: string[] }) {
   223→   223→  const [currentPhraseIndex, setCurrentPhraseIndex] = useState(0)
   224→   224→  const [currentText, setCurrentText] = useState('')
   225→   225→  const [isDeleting, setIsDeleting] = useState(false)
   226→   226→
   227→   227→  useEffect(() => {
   228→   228→    const phrase = phrases[currentPhraseIndex]
   229→   229→    if (!isDeleting) {
   230→   230→      if (currentText.length < phrase.length) {
   231→   231→        const timeout = setTimeout(() => {
   232→   232→          setCurrentText(phrase.slice(0, currentText.length + 1))
   233→   233→        }, 50)
   234→   234→        return () => clearTimeout(timeout)
   235→   235→      } else {
   236→   236→        const timeout = setTimeout(() => {
   237→   237→          setIsDeleting(true)
   238→   238→        }, 2000)
   239→   239→        return () => clearTimeout(timeout)
   240→   240→      }
   241→   241→    } else {
   242→   242→      if (currentText.length > 0) {
   243→   243→        const timeout = setTimeout(() => {
   244→   244→          setCurrentText(currentText.slice(0, -1))
   245→   245→        }, 50)
   246→   246→        return () => clearTimeout(timeout)
   247→   247→      } else {
   248→   248→        const timeout = setTimeout(() => {
   249→   249→          setIsDeleting(false)
   250→   250→          setCurrentPhraseIndex((prev) => (prev + 1) % phrases.length)
   251→   251→        }, 50)
   252→   252→        return () => clearTimeout(timeout)
   253→   253→      }
   254→   254→    }
   255→   255→  }, [currentText, isDeleting, currentPhraseIndex, phrases])
   256→   256→
   257→   257→  return (
   258→   258→    <span className="text-emerald-300 font-bold text-xl">
   259→   259→      {currentText}
   260→   260→      <span className="animate-pulse">|</span>
   261→   261→    </span>
   262→   262→  )
   263→   263→}

   273→   273→      ref={ref}
   274→   274→      initial={{ opacity: 0, y: 40 }}
   275→   275→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
   276→   276→      transition={{ duration: 0.6, ease: 'easeOut' }}
   277→   277→      className={className}
   278→   278→    >
   279→   279→      {children}
   280→   280→    </motion.div>
   281→   281→  )
   282→   282→}
