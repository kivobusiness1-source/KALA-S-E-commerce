   150→function OrderProgressIndicator({ step }: { step: 1 | 2 | 3 }) {
   151→  const steps = [
   152→    { num: 1, label: 'Informations' },
   153→    { num: 2, label: 'Livraison' },
   154→    { num: 3, label: 'Confirmation' },
   155→  ]

   157→    <div className="flex items-center justify-between mb-6">
   158→      {steps.map((s, i) => (
   159→        <div key={s.num} className="flex items-center flex-1 last:flex-none">
   160→          <div className="flex flex-col items-center gap-1.5">
   161→            <div
   162→              className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-semibold transition-colors duration-300 ${
   163→                step >= s.num
   164→                  ? 'bg-emerald-600 text-white shadow-md shadow-emerald-600/25'
   165→                  : 'bg-gray-100 text-gray-400'
   166→              }`}
   167→            >
   168→              {step > s.num ? <Check className="w-4 h-4" /> : s.num}
   169→            </div>
   170→            <span className={`text-[11px] font-medium ${step >= s.num ? 'text-emerald-700' : 'text-gray-400'}`}>
   171→              {s.label}
   172→            </span>
   173→          </div>
   174→          {i < steps.length - 1 && (
   175→            <div className={`flex-1 h-0.5 mx-3 rounded-full transition-colors duration-300 ${step > s.num ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   176→          )}
   177→        </div>
   178→      ))}
   179→    </div>
   180→  )
   181→}
